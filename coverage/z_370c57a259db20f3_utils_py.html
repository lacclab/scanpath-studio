79@st.cache_data(show_spinner="Building trial list…") 

80def _build_combo_options_cached( 

81 _fixations: pd.DataFrame, 

82 composite_cols: Tuple[str, ...], 

83 cache_key, 

84) -> Tuple[pd.DataFrame, list[str], Dict[str, Tuple[str, str]]]: 

85 fixations = _fixations 

86 trial_col = ( 

87 "unique_trial_id" if "unique_trial_id" in fixations.columns else "trial_id" 

88 ) 

89 # The text/passage column is optional. Normalized frames carry a text_id (it 

90 # falls back to trial_id when no text is mapped), but a frame may arrive with 

91 # only the source name (e.g. unique_paragraph_id — the pre-rename text id, and 

92 # which can also be a composite-trial component). Detect it via the same 

93 # priority list normalization uses, and *copy* it to text_id rather than 

94 # renaming so a shared composite component column survives for the picker. 

95 text_col = next( 

96 ( 

97 c 

98 for c in ( 

99 "unique_text_id", 

100 "text_id", 

101 "unique_paragraph_id", 

102 "paragraph_id", 

103 ) 

104 if c in fixations.columns 

105 ), 

106 None, 

107 ) 

108 combo_cols = ["participant_id", trial_col] 

109 if text_col is not None and text_col not in combo_cols: 

110 combo_cols.append(text_col) 

111 for col in ["unique_trial_id", "unique_text_id", "TRIAL_INDEX", "trial_index"]: 

112 if col in fixations.columns and col not in combo_cols: 

113 combo_cols.append(col) 

114 # Carry the composite trial id's component columns through, so the trial 

115 # picker can detect a composite id and cascade on identity (see select_trial). 

116 for col in composite_cols: 

117 if col in fixations.columns and col not in combo_cols: 

118 combo_cols.append(col) 

119 

120 # UX-24: preserve each trial's first appearance before the historical 

121 # participant/id sort. The visible pool may still default to Trial ID, but 

122 # the ⇅ menu can now reconstruct source-file order exactly. 

123 combos = fixations[combo_cols].drop_duplicates().copy() 

124 combos["_data_order"] = np.arange(len(combos), dtype=int) 

125 combos = combos.rename(columns={trial_col: "trial_id"}) 

126 if "text_id" not in combos.columns: 

127 combos["text_id"] = ( 

128 combos[text_col] if text_col is not None else combos["trial_id"] 

129 ) 

130 if trial_col == "unique_trial_id" and "unique_trial_id" not in combos.columns: 

131 combos["unique_trial_id"] = combos["trial_id"] 

132 if text_col == "unique_text_id" and "unique_text_id" not in combos.columns: 

133 combos["unique_text_id"] = combos["text_id"] 

134 sort_cols = ["participant_id"] 

135 if "TRIAL_INDEX" in combos.columns: 

136 sort_cols.append("TRIAL_INDEX") 

137 elif "trial_index" in combos.columns: 

138 sort_cols.append("trial_index") 

139 sort_cols.append("trial_id") 

140 combos = combos.sort_values(sort_cols) 

141 

142 combo_labels = [ 

143 f"{row.participant_id} / {row.trial_id} · {row.text_id}" 

144 for row in combos.itertuples() 

145 ] 

146 label_to_combo = dict( 

147 zip( 

148 combo_labels, 

149 combos[["participant_id", "trial_id"]].itertuples(index=False, name=None), 

150 ) 

151 ) 

152 return combos, combo_labels, label_to_combo

156def _trial_positions(_frame: pd.DataFrame, cache_key) -> Dict[Tuple[str, str], object]: 

157 """Map ``(participant_id, trial_id)`` → positional row indices. 

158 

159 Built once per frame (cached on its fingerprint) so extracting a single 

160 trial is an O(trial) ``iloc`` rather than an O(corpus) boolean mask on every 

161 rerun — and shared across the tabs, which all slice the same filtered frames. 

162 """ 

163 if _frame is None or _frame.empty: 

164 return {} 

165 grouped = _frame.groupby(["participant_id", "trial_id"], sort=False).indices 

166 # Normalise keys to (str, str) so lookups match the picker's string values. 

167 return {(str(p), str(t)): idx for (p, t), idx in grouped.items()}

170def extract_trial(frame: pd.DataFrame, participant_id, trial_id) -> pd.DataFrame: 

171 """Rows of one (participant, trial), sliced via the cached position index. 

172 

173 Equivalent to ``frame[(frame.participant_id == p) & (frame.trial_id == t)]`` 

174 but O(trial) instead of O(corpus) once the index is built — the per-rerun win 

175 on large datasets, where every tab extracts the selected trial.""" 

176 if frame is None or getattr(frame, "empty", True): 

177 return frame 

178 positions = _trial_positions(frame, cache_key=frame_fingerprint(frame)) 

179 pos = positions.get((str(participant_id), str(trial_id))) 

180 if pos is None or len(pos) == 0: 

181 return frame.iloc[0:0] 

182 return frame.iloc[pos]

289def _trial_sort_column_allowed(column: object) -> bool: 

290 """Whether ``column`` can be discovered as generic trial metadata.""" 

291 name = str(column) 

292 lower = name.lower() 

293 if name.startswith("_") or lower in _TRIAL_SORT_EXCLUDED_COLS: 

294 return False 

295 if lower.endswith(_TRIAL_SORT_GEOMETRY_SUFFIXES): 

296 return False 

297 return not any(part in lower for part in _TRIAL_SORT_PRIVATE_NAME_PARTS)

300def _effective_trial_field( 

301 frame: Optional[pd.DataFrame], trial_field: str, picker_ids: set[str] 

302) -> Optional[str]: 

303 """Find the frame column that names the picker's effective trial ids.""" 

304 if frame is None or frame.empty: 

305 return None 

306 for field in (trial_field, "unique_trial_id", "trial_id"): 

307 if field not in frame.columns: 

308 continue 

309 values = set(frame[field].dropna().astype(str).unique()) 

310 if values & picker_ids: 

311 return field 

312 return None

315def _is_missing_scalar(value) -> bool: 

316 try: 

317 missing = pd.isna(value) 

318 except (TypeError, ValueError): 

319 return False 

320 return bool(missing) if isinstance(missing, (bool, np.bool_)) else False

323def _same_sort_value(left, right) -> bool: 

324 """Scalar equality for reconciling the words and fixations tables.""" 

325 if _is_missing_scalar(left) or _is_missing_scalar(right): 

326 # Partial missingness is not a conflict; the table carrying a value wins. 

327 return True 

328 try: 

329 equal = left == right 

330 except (TypeError, ValueError): 

331 return False 

332 return bool(equal) if isinstance(equal, (bool, np.bool_)) else False

335def _looks_like_free_text(series: pd.Series) -> bool: 

336 """Reject prose-like cells while retaining ordinary categorical metadata.""" 

337 values = [str(v).strip() for v in series if not _is_missing_scalar(v)] 

338 return bool(values) and any(len(v) > 200 or len(v.split()) > 24 for v in values)

341def _trial_level_columns_from_frame( 

342 frame: Optional[pd.DataFrame], 

343 trial_field: str, 

344 picker_ids: set[str], 

345 participants: set[str], 

346) -> Dict[str, pd.Series]: 

347 """Discover one scalar value per active trial directly from one source table. 

348 

349 Grouping includes participant identity, preventing repeated plain trial ids 

350 from being merged before the active picker scope is applied. The returned 

351 Series uses the picker's effective id because that is what 

352 ``sort_trial_options`` consumes. 

353 """ 

354 identity = _effective_trial_field(frame, trial_field, picker_ids) 

355 if frame is None or frame.empty or identity is None: 

356 return {} 

357 

358 scoped = frame 

359 if participants and "participant_id" in scoped.columns: 

360 scoped = scoped[scoped["participant_id"].astype(str).isin(participants)] 

361 scoped = scoped[scoped[identity].astype(str).isin(picker_ids)] 

362 if scoped.empty: 

363 return {} 

364 

365 group_cols = [identity] 

366 if "participant_id" in scoped.columns and identity != "participant_id": 

367 group_cols.insert(0, "participant_id") 

368 grouped = scoped.groupby(group_cols, sort=False, dropna=False) 

369 discovered: Dict[str, pd.Series] = {} 

370 for col in scoped.columns: 

371 if col == identity or not _trial_sort_column_allowed(col): 

372 continue 

373 try: 

374 if (grouped[col].nunique(dropna=False) > 1).any(): 

375 continue 

376 if col in group_cols: 

377 values = scoped[group_cols].drop_duplicates().copy() 

378 else: 

379 values = grouped[col].agg(lambda cells: cells.iloc[0]).reset_index() 

380 # If the same effective id survives for multiple participants, it is 

381 # usable only when those rows agree. A participant-narrowed picker 

382 # naturally has one row here; a global ambiguous picker is not 

383 # allowed to choose one participant silently. 

384 by_id = values.groupby(values[identity].astype(str), sort=False)[col] 

385 if (by_id.nunique(dropna=False) > 1).any(): 

386 continue 

387 deduped = values.drop_duplicates(subset=[identity]) 

388 except (TypeError, ValueError): 

389 # Nested/list-like event payloads are not sortable scalar metadata. 

390 continue 

391 series = pd.Series( 

392 deduped[col].to_numpy(), 

393 index=deduped[identity].astype(str).to_numpy(), 

394 ) 

395 if series.dropna().empty or _looks_like_free_text(series): 

396 continue 

397 discovered[str(col)] = series 

398 return discovered

401def _merge_trial_level_sources( 

402 sources: Iterable[Dict[str, pd.Series]], 

403) -> Dict[str, pd.Series]: 

404 """Merge compatible metadata sources; omit cross-table disagreements.""" 

405 by_column: Dict[str, list[pd.Series]] = {} 

406 for source in sources: 

407 for col, series in source.items(): 

408 by_column.setdefault(col, []).append(series) 

409 

410 merged: Dict[str, pd.Series] = {} 

411 for col, series_list in by_column.items(): 

412 combined = pd.Series(dtype=object) 

413 conflict = False 

414 for series in series_list: 

415 current = series.copy() 

416 current.index = current.index.astype(str) 

417 for trial_id in combined.index.intersection(current.index): 

418 if not _same_sort_value(combined[trial_id], current[trial_id]): 

419 conflict = True 

420 break 

421 if conflict: 

422 break 

423 combined = combined.combine_first(current) 

424 if not conflict and not combined.empty: 

425 merged[col] = combined 

426 return merged

482def _per_trial_stat(frame: pd.DataFrame, trial_field: str, how: str) -> pd.Series: 

483 """One computed stat per trial id, as a Series indexed by that id.""" 

484 if frame is None or frame.empty or trial_field not in frame.columns: 

485 return pd.Series(dtype=float) 

486 grouped = frame.groupby(frame[trial_field].astype(str), sort=False) 

487 if how == "size": 

488 return grouped.size().astype(float) 

489 if how == "timestamp_min": 

490 if "timestamp_ms" not in frame.columns: 

491 return pd.Series(dtype=float) 

492 return pd.to_numeric(grouped["timestamp_ms"].min(), errors="coerce").astype( 

493 float 

494 ) 

495 if "duration_ms" not in frame.columns: 

496 return pd.Series(dtype=float) 

497 durations = grouped["duration_ms"].agg("sum" if "sum" in how else "mean") 

498 return (durations / 1000.0) if how.endswith("_s") else durations.astype(float)

501def trial_sort_keys( 

502 combos: pd.DataFrame, 

503 trial_field: str, 

504 *, 

505 words: Optional[pd.DataFrame] = None, 

506 fixations: Optional[pd.DataFrame] = None, 

507) -> Dict[str, pd.Series]: 

508 """Available sort keys (UX-10): label → Series indexed by trial id. 

509 

510 Offers a computed stat only when the frame it needs is present, and a column 

511 only when it is actually trial-level in the active participant-scoped words, 

512 fixations, or combo frame. This deliberately discovers metadata before the 

513 lossy combo projection can discard it. 

514 """ 

515 keys: Dict[str, pd.Series] = {} 

516 # This rank was captured before build_combo_options' canonical sort. 

517 if ( 

518 combos is not None 

519 and not combos.empty 

520 and trial_field in combos.columns 

521 and "_data_order" in combos.columns 

522 ): 

523 deduped = combos.drop_duplicates(subset=[trial_field]) 

524 keys["Data order"] = pd.Series( 

525 deduped["_data_order"].to_numpy(), 

526 index=deduped[trial_field].astype(str).to_numpy(), 

527 ) 

528 for label, (which, how) in _TRIAL_SORT_STATS.items(): 

529 frame = fixations if which == "fixations" else words 

530 picker_ids = ( 

531 set(combos[trial_field].dropna().astype(str).unique()) 

532 if combos is not None and not combos.empty and trial_field in combos.columns 

533 else set() 

534 ) 

535 field = _effective_trial_field(frame, trial_field, picker_ids) 

536 series = _per_trial_stat(frame, field or trial_field, how) 

537 if not series.empty: 

538 keys[label] = series 

539 if combos is None or combos.empty or trial_field not in combos.columns: 

540 return keys 

541 discovered = _trial_level_sort_columns(combos, trial_field, words, fixations) 

542 ordered_cols = [c for c in _TRIAL_SORT_PREFERRED_COLS if c in discovered] 

543 ordered_cols.extend(sorted(set(discovered) - set(ordered_cols), key=str.casefold)) 

544 for col in ordered_cols: 

545 if col == trial_field: 

546 continue 

547 keys[col.replace("_", " ").capitalize()] = discovered[col] 

548 return keys

551def sort_trial_options( 

552 options: List[str], 

553 key_series: Optional[pd.Series], 

554 *, 

555 descending: bool = False, 

556) -> List[str]: 

557 """Order ``options`` (trial ids) by ``key_series``, ties broken by id. 

558 

559 Trials the key doesn't cover sort last regardless of direction — an unranked 

560 trial is missing information, not an extreme value, so it shouldn't lead. 

561 """ 

562 if key_series is None or key_series.empty: 

563 return sorted(options) 

564 lookup = key_series.to_dict() 

565 ranked = [o for o in options if o in lookup and pd.notna(lookup[o])] 

566 unranked = sorted(o for o in options if o not in ranked) 

567 ranked.sort(key=lambda o: (_sort_scalar(lookup[o]), o), reverse=descending) 

568 return ranked + unranked

571def _sort_scalar(value): 

572 """A comparable key for a cell that may be numeric, boolean or text.""" 

573 if isinstance(value, bool): 

574 return (0, float(value)) 

575 try: 

576 return (0, float(value)) 

577 except (TypeError, ValueError): 

578 return (1, str(value))

581def format_sort_value(value) -> str: 

582 """A sort key's value, short enough to ride along in a picker option. 

583 

584 Sorting the pool is only useful if you can *see* what you sorted by — an 

585 ordering with the ordering key hidden just looks shuffled. Integers keep a 

586 thousands separator, floats get one decimal, booleans read Yes/No. 

587 """ 

588 if value is None or (isinstance(value, float) and pd.isna(value)): 

589 return "—" 

590 if isinstance(value, (bool, np.bool_)): 

591 return "Yes" if value else "No" 

592 if isinstance(value, (int, float, np.integer, np.floating)): 

593 number = float(value) 

594 return f"{number:,.0f}" if number == int(number) else f"{number:,.1f}" 

595 return str(value)

600 

601 A per-page trial id reads cleanly — ``Lit_Alchemist_4__page_07`` → 

602 ``Lit_Alchemist_4 · page 7`` (the id stays zero-padded so it sorts 

603 numerically; only the display drops the padding). Any other id passes 

604 through unchanged, so this is a no-op for every other corpus. MultiplEYE 

605 used to be the one producing those ids; since DATA-24 its pages are screens 

606 inside one trial, so this is now generic dressing for whatever ships them.""" 

607 text = str(trial_id) 

608 stim, sep, page = text.rpartition("__page_") 

609 if sep and page.isdigit(): 

610 return f"{stim} · page {int(page)}" 

611 return text

614def _render_trial_sort_popover( 

615 host, 

616 combos: pd.DataFrame, 

617 trial_field: str, 

618 key_prefix: str, 

619 *, 

620 words: Optional[pd.DataFrame], 

621 fixations: Optional[pd.DataFrame], 

622) -> Tuple[Optional[pd.Series], bool, str]: 

623 """The ⇅ sort control beside the trial picker (UX-10). 

624 

625 Lives in a popover rather than inline: the picker row is already a selectbox, 

626 a slider and two step buttons wide, and sorting is a "set it once" choice, not 

627 a per-trial one. Returns ``(key_series, descending, choice)`` for 

628 :func:`sort_trial_options` — ``(None, False, TRIAL_SORT_DEFAULT)`` for the 

629 default id order. The chosen key's *name* comes back too, because the picker 

630 labels the ordering it's showing. 

631 """ 

632 keys = trial_sort_keys(combos, trial_field, words=words, fixations=fixations) 

633 if not keys: 

634 return None, False, TRIAL_SORT_DEFAULT 

635 options = [TRIAL_SORT_DEFAULT, *keys] 

636 state_key = f"{key_prefix}_trial_sort" 

637 if st.session_state.get(state_key) not in options: 

638 st.session_state[state_key] = TRIAL_SORT_DEFAULT 

639 with host.popover("⇅", width="content", help="Sort the trial list"): 

640 choice = st.selectbox( 

641 "Sort trials by", 

642 options=options, 

643 key=state_key, 

644 help="Reorder the picker so outliers surface without scrolling — by a " 

645 "computed stat (fixation count, reading time) or a reader / text / " 

646 "condition property.", 

647 ) 

648 descending = st.checkbox("Descending", key=f"{key_prefix}_trial_sort_desc") 

649 if choice == TRIAL_SORT_DEFAULT: 

650 return None, False, TRIAL_SORT_DEFAULT 

651 return keys[choice], bool(descending), choice

654# --- CMP-13: one ◀ ▶ that advances both compared trials ---------------------- 

655# The two pickers are built in different modules (A here, B in `tabs.py`), so the 

656# linked step is a callback on one side writing the *other* side's selection. It 

657# needs the other list, which is why each picker publishes what it just rendered. 

658# Both directions clamp independently: per the settled call, a side that has run 

659# out simply stays put while the other keeps stepping. 

660 

661#: The ⚙️ Compare options checkbox that arms the link. UI-only — a navigation 

662#: control, not a render setting, so it is deliberately not on the share link or 

663#: in a saved config (same call as ``share_identity_mode``). 

664COMPARE_STEP_LINK_KEY = "single_compare_step_linked" 

665 

666#: Scanpath B's canonical selection (a *label*; see the snapshot note below). 

667COMPARE_TRIAL_KEY = "single_compare_trial" 

668 

669#: What the *Compare To* picker last rendered: ``(label, participant, trial)`` 

670#: per candidate, in display order. 

671COMPARE_OPTIONS_SNAPSHOT_KEY = "_compare_options_snapshot"

674def trial_options_snapshot_key(key_prefix: str) -> str: 

675 """Session key holding the trial picker's options as it last rendered them.""" 

676 return f"_{key_prefix}_trial_options" if key_prefix else "_trial_options"

679def compare_step_linked() -> bool: 

680 """True when ◀ ▶ should advance scanpath A **and** B (CMP-13). 

681 

682 Both halves matter: the checkbox only exists while compare mode is on, and 

683 Streamlit drops an unrendered widget's key, so a stale ``True`` must not 

684 quietly steer the main picker once the user has left compare mode. 

685 """ 

686 return bool( 

687 st.session_state.get("single_compare_toggle") 

688 and st.session_state.get(COMPARE_STEP_LINK_KEY)

692def step_within(options: List[str], state_key: str, delta: int) -> Optional[int]: 

693 """Move ``state_key``'s selection ``delta`` places within ``options``. 

694 

695 Clamped to the ends, and clamped *independently* of any other picker — the 

696 linked step is "advance both", not "keep them aligned": the two pools have 

697 different sizes (B excludes A, and a cross-dataset B is another corpus 

698 entirely), so their indices carry no shared meaning. 

699 

700 Returns the new index, or ``None`` when there was nothing to step. 

701 """ 

702 opts = list(options or []) 

703 if not opts: 

704 return None 

705 try: 

706 pos = opts.index(st.session_state.get(state_key)) 

707 except ValueError: 

708 pos = 0 

709 new_pos = max(0, min(pos + delta, len(opts) - 1)) 

710 st.session_state[state_key] = opts[new_pos] 

711 return new_pos

713 

714def at_list_end(options: List[str], state_key: str, delta: int) -> bool: 

715 """True when ``state_key``'s selection cannot move ``delta`` within ``options``. 

716 

717 Used to decide whether a step button is dead. An unknown list answers 

718 **False** so the button stays live: a click that turns out to be a no-op is a 

719 better failure than a button greyed out while the other side could still move. 

720 """ 

721 opts = list(options or []) 

722 if not opts: 

723 return False 

724 try: 

725 pos = opts.index(st.session_state.get(state_key)) 

726 except ValueError: 

727 return False 

728 return not (0 <= pos + delta < len(opts))

731def step_linked_compare(delta: int) -> None: 

732 """Advance scanpath **B** by ``delta``, resolved against the list A last saw. 

733 

734 Written as an *identity* rather than an index or a label, because both are 

735 unstable across this step: ``build_comparison_options`` builds B's pool 

736 relative to A (📄 same-text first, then 👤 same-participant, A itself 

737 excluded), so once A moves, B's list is re-ordered *and* re-labelled — the 

738 same trial can gain or lose its 📄 marker. Parking the identity in the same 

739 pending slot the ``?compare=`` deep link uses lets the rebuilt picker re-find 

740 the trial the user was actually looking at. 

741 """ 

742 from .session_keys import PENDING_COMPARE_STATE_KEY 

743 

744 snapshot = list(st.session_state.get(COMPARE_OPTIONS_SNAPSHOT_KEY) or []) 

745 if not snapshot: 

746 return 

747 labels = [row[0] for row in snapshot] 

748 try: 

749 pos = labels.index(st.session_state.get(COMPARE_TRIAL_KEY)) 

750 except ValueError: 

751 pos = 0 

752 _, participant, trial = snapshot[max(0, min(pos + delta, len(snapshot) - 1))] 

753 st.session_state[PENDING_COMPARE_STATE_KEY] = { 

754 "participant_id": participant, 

755 "trial_id": trial, 

756 }

759def _select_trial_none_mode( 

760 combos: pd.DataFrame, 

761 trial_field: str, 

762 text_field: str, 

763 key_prefix: str, 

764 picker_host=None, 

765 *, 

766 words: Optional[pd.DataFrame] = None, 

767 fixations: Optional[pd.DataFrame] = None, 

768) -> Tuple[Optional[str], Optional[str], Optional[str]]: 

769 """The trial picker: a **selectbox + scrubbing slider + ◀ ▶ step buttons + a ⇅ 

770 sort popover**, all on one row. The slider thumb shows ``index/TOTAL · id`` 

771 (index first). The pool is narrowed upstream (the "Narrow by" multiselects + 

772 the "More" filters), so this just picks one trial from it and orders it. 

773 

774 Creates its own row of columns, so call it where columns are allowed (the 

775 Scanpath/Corpus body), not nested inside another column. ``picker_host`` 

776 (when given) is the container to render into; defaults to the current one. 

777 ``words`` / ``fixations`` (optional) unlock the computed sort keys (UX-10); 

778 without them only column-based orderings are offered.""" 

779 host = picker_host if picker_host is not None else st 

780 available_trials = combos.drop_duplicates(subset=[trial_field]) 

781 trial_options = sorted(available_trials[trial_field].dropna().astype(str).unique()) 

782 if not trial_options: 

783 st.warning("No trials available after filtering.") 

784 st.stop() 

785 

786 # Trial id → participant, so the annotation markers (UX-6) can be looked up per 

787 # option (annotations are keyed by (participant, trial)). Mirrors the selection 

788 # below, which resolves the participant the same way (first matching row). 

789 trial_to_pid = { 

790 str(r[trial_field]): r["participant_id"] 

791 for r in available_trials.to_dict("records") 

792 } 

793 

794 # Populated once the ⇅ popover has rendered (below), and read by the option 

795 # labels — so an active ordering is *visible* in the picker itself rather than 

796 # only inside the popover that set it. 

797 sort_values: Dict[str, str] = {} 

798 

799 def _option_label(value: str) -> str: 

800 marks = annotation_markers(trial_to_pid.get(value), value) 

801 base = _trial_display_label(value) 

802 label = f"{marks} {base}" if marks else base 

803 shown = sort_values.get(value) 

804 return f"{label} · {shown}" if shown else label 

805 

806 n_trials = len(trial_options) 

807 picker_label = "**Select Trial**" 

808 trial_id_key = f"{key_prefix}_trial_id" if key_prefix else None 

809 slider_key = f"{key_prefix}_trial_pos" if key_prefix else "trial_pos" 

810 

811 # The selectbox (`*_trial_id`) is the canonical selection — the deep-link / 

812 # Save-&-restore code seeds it (`_restore_selection`). The slider mirrors it 

813 # and ◀ ▶ step it; all stay in sync via the trial id. 

814 current_label = st.session_state.get(trial_id_key) if trial_id_key else None 

815 if current_label not in trial_options: 

816 current_label = trial_options[0] 

817 if trial_id_key: 

818 st.session_state[trial_id_key] = current_label 

819 

820 idx_of = {opt: i for i, opt in enumerate(trial_options)} 

821 

822 if n_trials > 1: 

823 # Mirror the slider to the current selection BEFORE it renders, so picking 

824 # a trial in the dropdown moves the slider too; the drag callback writes 

825 # the selectbox key, reconciling next run. 

826 st.session_state[slider_key] = current_label 

827 

828 def _on_trial_slider() -> None: 

829 if trial_id_key: 

830 st.session_state[trial_id_key] = st.session_state[slider_key] 

831 

832 def _step_trial(delta: int) -> None: 

833 # ◀ / ▶ : move the canonical selection one trial earlier/later. 

834 if not trial_id_key: 

835 return 

836 step_within(trial_options, trial_id_key, delta) 

837 # CMP-13: while the link is armed, the same ±1 also moves scanpath B. 

838 if compare_step_linked(): 

839 step_linked_compare(delta) 

840 

841 def _slider_label(value: str) -> str: 

842 # index/TOTAL first, then the id — the slider doubles as the counter, 

843 # so a separate "Trial X / N" caption is redundant. 

844 return f"{idx_of.get(value, 0) + 1}/{n_trials} · {_option_label(value)}" 

845 

846 # One row: [selectbox] [slider] [◀ ▶ ⇅] — the three triggers share ONE 

847 # trailing column as a `railbtn_*` cluster (UX-27), which styles.py packs 

848 # right at a uniform 3px spacing. A column each put a full gutter between 

849 # them, so a prev/next *pair* didn't read as a pair and the group didn't 

850 # line up with the **More** row above or the chip row below. 

851 sel_col, slider_col, trail_col = host.columns( 

852 SELECTOR_ROW_GRID, vertical_alignment="bottom" 

853 ) 

854 trail = trail_col.container(key=f"railbtn_{key_prefix}_trail") 

855 # Created in display order (◀ ▶ then ⇅) but filled out of order: the sort 

856 # popover has to render first, because the order it returns is what the 

857 # selectbox, the slider and the ◀ ▶ steps all walk. 

858 step_col = trail.container(key=f"railbtn_{key_prefix}_step") 

859 sort_col = trail.container(key=f"railbtn_{key_prefix}_sort") 

860 # UX-10: order the pool *before* the widgets read `trial_options`, so the 

861 # selectbox, the slider and the ◀ ▶ steps all walk the same order. The 

862 # canonical selection is a trial *id*, so re-sorting never changes which 

863 # trial is selected — only where it sits in the list. 

864 # UX-27: each of the three step/sort triggers goes in a `railbtn_*` 

865 # container so styles.py can give the whole cluster above the plot one 

866 # button shape — these were square (each `width="stretch"` inside its own 

867 # narrow column) beside pill-shaped labelled buttons on the rows above 

868 # and below. 

869 sort_key, sort_desc, sort_choice = _render_trial_sort_popover( 

870 sort_col, 

871 combos, 

872 trial_field, 

873 key_prefix, 

874 words=words, 

875 fixations=fixations, 

876 ) 

877 if sort_key is not None: 

878 trial_options = sort_trial_options( 

879 trial_options, sort_key, descending=sort_desc 

880 ) 

881 idx_of = {opt: i for i, opt in enumerate(trial_options)} 

882 lookup = sort_key.to_dict() 

883 sort_values.update( 

884 {opt: format_sort_value(lookup.get(opt)) for opt in trial_options} 

885 ) 

886 picker_label = ( 

887 f"**Select Trial** · by {sort_choice} {'↓' if sort_desc else '↑'}" 

888 ) 

889 current_idx = trial_options.index(current_label) 

890 else: 

891 sel_col = host 

892 

893 # CMP-13: publish the list as rendered (post-sort), so the *Compare To* 

894 # picker's linked ◀ ▶ can step this picker without rebuilding its ordering. 

895 if trial_id_key: 

896 st.session_state[trial_options_snapshot_key(key_prefix)] = list(trial_options) 

897 

898 # The label is shown so its help "?" icon (the type-to-search hint) is visible 

899 # — a collapsed label hides it. 

900 selected_trial_label = sel_col.selectbox( 

901 picker_label, 

902 options=trial_options, 

903 key=trial_id_key, 

904 format_func=_option_label, 

905 help="💡 Click, then start typing to narrow down the trial list. " 

906 "★ favorite · 🏷️ tagged · 📝 has notes. When a sort key is active, each " 

907 "option ends with that trial's value for it.", 

908 ) 

909 

910 if n_trials > 1: 

911 with slider_col: 

912 st.select_slider( 

913 "Trial", 

914 options=trial_options, 

915 key=slider_key, 

916 on_change=_on_trial_slider, 

917 help=f"Scrub through the {n_trials} trials (index/total · id, " 

918 "plus the sort value when one is active); the dropdown jumps to " 

919 "a specific id.", 

920 label_visibility="collapsed", 

921 format_func=_slider_label, 

922 ) 

923 # Both step buttons in the keyed container reserved above, which styles.py 

924 # lays out as a flex ROW (a Streamlit vertical block stacks its children 

925 # by default). 

926 steps = step_col 

927 # CMP-13: linked, a button stays live until BOTH sides have run out — a 

928 # side at the end of its own list just stays put while the other keeps 

929 # stepping. `at_list_end` answers False for a list it can't see, so the 

930 # worst case is a click that moves only one scanpath. 

931 linked = compare_step_linked() 

932 compare_snapshot = ( 

933 [ 

934 row[0] 

935 for row in (st.session_state.get(COMPARE_OPTIONS_SNAPSHOT_KEY) or []) 

936 ] 

937 if linked 

938 else [] 

939 ) 

940 step_help = " Linked: also steps the compared trial." if linked else "" 

941 steps.button( 

942 "◀", 

943 key=f"{key_prefix}_prev_trial" if key_prefix else "prev_trial", 

944 on_click=_step_trial, 

945 args=(-1,), 

946 disabled=current_idx == 0 

947 and (not linked or at_list_end(compare_snapshot, COMPARE_TRIAL_KEY, -1)), 

948 help="Previous trial." + step_help, 

949 ) 

950 steps.button( 

951 "▶", 

952 key=f"{key_prefix}_next_trial" if key_prefix else "next_trial", 

953 on_click=_step_trial, 

954 args=(1,), 

955 disabled=current_idx == n_trials - 1 

956 and (not linked or at_list_end(compare_snapshot, COMPARE_TRIAL_KEY, 1)), 

957 help="Next trial." + step_help, 

958 ) 

959 

960 if not selected_trial_label: 

961 return None, None, None 

962 

963 chosen = available_trials[ 

964 available_trials[trial_field].astype(str) == selected_trial_label 

965 ].iloc[0] 

966 selected_text = str(chosen[text_field]) if text_field in chosen.index else None 

967 return chosen["participant_id"], chosen["trial_id"], selected_text

970def select_trial( 

971 combos: pd.DataFrame, 

972 key_prefix: str = "", 

973 picker_host=None, 

974 *, 

975 words: Optional[pd.DataFrame] = None, 

976 fixations: Optional[pd.DataFrame] = None, 

977) -> Tuple[Optional[str], Optional[str], str, Optional[str]]: 

978 """Pick a specific trial from the (already-narrowed) pool. 

979 

980 There are no Browse-by modes anymore, and no per-mapping variants either: the 

981 pool is narrowed by the inline **Narrow by** Text / Participant multiselects + 

982 the **More** filters (``controls.render_narrow_by`` / 

983 ``render_trial_filters``), and this always picks one trial via the same 

984 selectbox + slider + ◀ ▶ arrows — whatever the Trial ID mapping looks like. 

985 

986 BUG-23: a composite trial id (built from several mapped columns) used to get a 

987 picker of its own — first one selector per mapped component, then a Participant 

988 → Text cascade. Both made the *shape of the mapping* visible in the UI, and 

989 neither offered the slider or the step buttons, so stepping through trials 

990 worked on some datasets and not others. Participant and Text are what **Narrow 

991 by** is for; the composite flag now only tells the chip strip to spell the 

992 joined id out (``tabs._render_trial_condition_chips``). 

993 

994 ``picker_host`` is the container to render into (defaults to the current one); 

995 the picker builds its own row of columns, so call it where columns are allowed. 

996 

997 ``words`` / ``fixations`` (optional) are the frames ``combos`` was built from; 

998 passing them unlocks the UX-10 ⇅ sort popover's computed keys (fixation count, 

999 reading time). Without them the sort still offers the column-based orderings. 

1000 

1001 Returns: 

1002 Tuple of (participant_id, trial_id, selection_mode, selected_text). 

1003 ``selection_mode`` is always ``"Trial"`` (kept for the comparison-options 

1004 builder, which still supports the other modes when called directly). 

1005 """ 

1006 if combos.empty: 

1007 st.warning("No trials available after filtering.") 

1008 st.stop() 

1009 

1010 trial_field = ( 

1011 "unique_trial_id" if "unique_trial_id" in combos.columns else "trial_id" 

1012 ) 

1013 text_field = "unique_text_id" if "unique_text_id" in combos.columns else "text_id" 

1014 

1015 participant, trial, text = _select_trial_none_mode( 

1016 combos, 

1017 trial_field, 

1018 text_field, 

1019 key_prefix, 

1020 picker_host=picker_host, 

1021 words=words, 

1022 fixations=fixations, 

1023 ) 

1024 

1025 return participant, trial, "Trial", text

1060def gather_trial_metadata( 

1061 trial_words: pd.DataFrame, trial_fixations: pd.DataFrame, fields: Iterable[str] 

1062) -> pd.DataFrame: 

1063 """Gather metadata for selected fields from words and fixations.""" 

1064 rows = [] 

1065 for field in fields: 

1066 if field in trial_words.columns: 

1067 series = pd.Series(trial_words[field]) 

1068 elif field in trial_fixations.columns: 

1069 series = pd.Series(trial_fixations[field]) 

1070 else: 

1071 continue 

1072 

1073 cleaned = series.dropna() 

1074 if cleaned.empty: 

1075 value = "—" 

1076 else: 

1077 unique_values = cleaned.unique() 

1078 if len(unique_values) == 1: 

1079 value = unique_values[0] 

1080 else: 

1081 numeric_series = pd.to_numeric(cleaned, errors="coerce") 

1082 numeric_values = numeric_series.dropna() 

1083 is_numeric = ( 

1084 not pd.api.types.is_bool_dtype(cleaned) 

1085 and ( 

1086 pd.api.types.is_numeric_dtype(cleaned) 

1087 or len(numeric_values) == len(cleaned) 

1088 ) 

1089 and not numeric_values.empty 

1090 ) 

1091 if is_numeric: 

1092 value = f"mean={numeric_values.mean():.2f}, std={numeric_values.std():.2f}" 

1093 else: 

1094 modes = cleaned.mode(dropna=True) 

1095 mode_value = modes.iloc[0] if not modes.empty else "—" 

1096 value = f"{mode_value} (mode, {len(unique_values)} unique)" 

1097 rows.append({"Field": field, "Value": value}) 

1098 

1099 df = pd.DataFrame(rows) 

1100 if not df.empty: 

1101 df["Value"] = df["Value"].astype(str) 

1102 return df

1151def friendly_trial_label( 

1152 participant_id: str, 

1153 trial_id: str, 

1154 text_id: Optional[str], 

1155 existing_labels: set[str], 

1156 prefix: str = "", 

1157) -> str: 

1158 """Create a short, de-duplicated label for comparison dropdowns/legends.""" 

1159 trial_str = str(trial_id) if trial_id is not None else "" 

1160 text_str = str(text_id) if text_id is not None else "" 

1161 text_str = text_str.strip() 

1162 trial_contains_text = text_str and text_str.lower() in trial_str.lower() 

1163 

1164 if text_str: 

1165 base = f"{text_str} · {participant_id}" 

1166 if not trial_contains_text: 

1167 base = f"{base} (trial {trial_str})" if trial_str else base 

1168 elif trial_str != text_str: 

1169 # Surface any trial_id suffix beyond the text id (e.g. a 

1170 # repeat-reading "_r2" tag added during normalization). Without 

1171 # this the primary and compare titles look identical when a 

1172 # participant re-read the same text. 

1173 extra = trial_str 

1174 if extra.lower().startswith(text_str.lower()): 

1175 extra = extra[len(text_str) :].lstrip("_- ") 

1176 if extra: 

1177 base = f"{text_str} ({extra}) · {participant_id}" 

1178 else: 

1179 base = f"{trial_str} · {participant_id}" if trial_str else participant_id 

1180 

1181 label = f"{prefix}{base}" 

1182 if label in existing_labels: 

1183 label = f"{prefix}{base} [{trial_str or 'trial'}]" 

1184 existing_labels.add(label) 

1185 return label

1188def build_comparison_options( 

1189 combos: pd.DataFrame, 

1190 selection_mode: str, 

1191 primary_participant: str, 

1192 primary_trial: str, 

1193 primary_text: Optional[str], 

1194 *, 

1195 cross_dataset: bool = False, 

1196) -> list[Tuple[str, str, str, str]]: 

1197 """Build a prioritized list of comparison-trial options. 

1198 

1199 Returns ``(participant_id, trial_id, label, markers)`` tuples, where 

1200 ``markers`` leads with the relation icons ``"📄"`` (same text) / ``"👤"`` (same 

1201 participant) and then the trial's annotation markers ``★`` (favorite) / ``🏷️`` 

1202 (tagged) / ``📝`` (noted) when present (UX-6), and ``label`` is 

1203 ``"<markers> <trial_id>"``. Ordered: same-text (📄) first, then same-participant 

1204 (👤), then the rest. A trial that is BOTH same-text and same-participant sorts 

1205 with the 📄 group (text-matches lead) and shows both markers. 

1206 

1207 ``cross_dataset`` (CMP-8 §5.1) says ``combos`` describes a *different* 

1208 dataset, and degrades the three id-based signals that would otherwise lie: 

1209 two corpora do not share readers, so 👤 never fires; the annotation store is 

1210 keyed on the **active** dataset's ids, so a foreign trial must not inherit 

1211 another reader's ★; and the primary trial is not in this pool, so a 

1212 coincidentally identical ``(participant, trial)`` is a real candidate rather 

1213 than the trial being compared. 📄 survives — a text id that matches across 

1214 corpora is exactly the pairing this feature exists for. 

1215 """ 

1216 text_field = "unique_text_id" if "unique_text_id" in combos.columns else "text_id" 

1217 uniq = combos.drop_duplicates(subset=["participant_id", "trial_id"]) 

1218 

1219 rows: list[dict] = [] 

1220 for row in uniq.itertuples(): 

1221 if not cross_dataset and (row.participant_id, row.trial_id) == ( 

1222 primary_participant, 

1223 primary_trial, 

1224 ): 

1225 continue 

1226 text_id = getattr(row, text_field, "") 

1227 same_text = bool(primary_text and str(text_id) == str(primary_text)) 

1228 same_participant = not cross_dataset and bool( 

1229 str(row.participant_id) == str(primary_participant) 

1230 ) 

1231 markers = ( 

1232 (SAME_TEXT_MARKER if same_text else "") 

1233 + (SAME_PARTICIPANT_MARKER if same_participant else "") 

1234 + ( 

1235 "" 

1236 if cross_dataset 

1237 else annotation_markers(row.participant_id, row.trial_id) 

1238 ) 

1239 ) 

1240 rows.append( 

1241 { 

1242 "participant_id": row.participant_id, 

1243 "trial_id": row.trial_id, 

1244 "same_text": same_text, 

1245 "same_participant": same_participant, 

1246 "markers": markers, 

1247 } 

1248 ) 

1249 

1250 # ★ group first, then 👤 group, then the rest. same_text is the primary sort 

1251 # key so a both-★-👤 trial leads the ★ group. Stable sort preserves combos 

1252 # order within a group. 

1253 rows.sort( 

1254 key=lambda r: (0 if r["same_text"] else 1, 0 if r["same_participant"] else 1) 

1255 ) 

1256 

1257 used_labels: set[str] = set() 

1258 options: list[Tuple[str, str, str, str]] = [] 

1259 for r in rows: 

1260 label = _compare_option_label( 

1261 r["participant_id"], r["trial_id"], r["markers"], used_labels 

1262 ) 

1263 options.append((r["participant_id"], r["trial_id"], label, r["markers"])) 

1264 return options

1281def qualify_for_compare(frame: pd.DataFrame, dataset: str) -> pd.DataFrame: 

1282 """A copy of ``frame`` whose ``participant_id`` is namespaced by ``dataset``. 

1283 

1284 Two corpora can hold the same ``(participant_id, trial_id)``, and 

1285 `plots.make_comparison_figure` slices its frame by exactly that pair — so an 

1286 unqualified merge would silently render *the wrong scanpath*, or two. 

1287 

1288 Only ever applied to the single-trial frames that feed the comparison 

1289 builder. Nothing the annotations, the export slug, the deep link or Corpus 

1290 Analysis reads goes through here: those key on the real ids, and must. 

1291 """ 

1292 if frame.empty: 

1293 return frame 

1294 out = frame.copy() 

1295 out["dataset"] = dataset 

1296 out["participant_id"] = ( 

1297 dataset + COMPARE_DATASET_SEP + out["participant_id"].astype(str) 

1298 ) 

1299 return out

1322def align_compare_columns( 

1323 a: pd.DataFrame, b: pd.DataFrame 

1324) -> Tuple[pd.DataFrame, pd.DataFrame, frozenset]: 

1325 """Reindex two frames onto their column **union**, and report shared numerics. 

1326 

1327 A bare ``pd.concat`` of frames with disjoint columns warns and churns dtypes 

1328 (int columns become float once the other frame's rows fill in as NaN), which 

1329 matters here because two corpora rarely ship the same measure set. Aligning 

1330 first keeps the concat quiet and the dtypes stable. 

1331 

1332 The third element is the intersection of *numeric* columns — the metrics a 

1333 cross-dataset figure may legitimately colour by (CMP-8 §5.4). A metric present 

1334 in only one corpus would colour one panel and blank the other. 

1335 """ 

1336 union = list(dict.fromkeys([*a.columns, *b.columns])) 

1337 a_aligned = a.reindex(columns=union) if list(a.columns) != union else a 

1338 b_aligned = b.reindex(columns=union) if list(b.columns) != union else b 

1339 shared = frozenset( 

1340 col 

1341 for col in set(a.columns) & set(b.columns) 

1342 if pd.api.types.is_numeric_dtype(a[col]) 

1343 and pd.api.types.is_numeric_dtype(b[col]) 

1344 ) 

1345 return a_aligned, b_aligned, shared
