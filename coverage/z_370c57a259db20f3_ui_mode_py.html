109def parse_mode(value: object) -> str | None: 

110 """The mode ``value`` names, or ``None`` when it is not a string naming one.""" 

111 if not isinstance(value, str): 

112 return None 

113 mode = value.strip().lower() 

114 return mode if mode in MODES else None

117def resolve_mode(param: object, cookie: object) -> str: 

118 """The link's mode, else the cookie's, else Essentials; junk values are skipped.""" 

119 return parse_mode(param) or parse_mode(cookie) or SIMPLE

122def seed_mode(session: MutableMapping, query_params: Mapping, cookies: Mapping) -> str: 

123 """Fix the session's mode on its first run and return it. 

124 

125 A mode already in the session wins: the header switch writes it, and a 

126 link's ``?ui=`` must not undo a choice made after the page opened. Past 

127 the link, the choice this session last wrote to the cookie 

128 (``COOKIE_MODE_KEY``, which survives starting over) beats ``cookies``, 

129 which is read once, when the page connects, and so misses that write. 

130 """ 

131 remembered = parse_mode(session.get(COOKIE_MODE_KEY)) or cookies.get(UI_MODE_COOKIE) 

132 session[UI_MODE] = parse_mode(session.get(UI_MODE)) or resolve_mode( 

133 query_params.get(PARAM_UI), remembered 

134 ) 

135 return session[UI_MODE]

138def seed_simple_start( 

139 session: MutableMapping, query_params: Mapping, *, restored: bool 

140) -> None: 

141 """Open a fresh Essentials session on the playing replay. 

142 

143 The replay is a starting value, not a lock: it applies only when the session 

144 starts in Essentials with no deep link and nothing restored, so a link, a saved 

145 configuration or a recovered session keeps what it carries, and Full 

146 keeps its static start. It runs once, on the first run, so switching to 

147 Essentials mid-session never turns the replay on. 

148 """ 

149 if session.get(START_DONE_KEY): 

150 return 

151 session[START_DONE_KEY] = True 

152 if ( 

153 session.get(UI_MODE) == SIMPLE 

154 and not has_link_params(query_params) 

155 and not restored 

156 ): 

157 session.setdefault(SINGLE_ANIMATE, True)

179def cookie_script(mode: str) -> str: 

180 """A same-origin script that remembers ``mode`` in a year-long cookie. 

181 

182 ``st.iframe`` embeds share the app's origin, so the parent document's 

183 ``cookie`` is writable from here, the only way to keep a preference 

184 browser-side without an account. 

185 """ 

186 value = f"{UI_MODE_COOKIE}={parse_mode(mode) or SIMPLE}; max-age={_COOKIE_MAX_AGE}" 

187 return ( 

188 "<script>window.parent.document.cookie = " 

189 f'"{value}; path=/; SameSite=Lax";</script>' 

190 )

198def mode_css(mode: str) -> str: 

199 """A ``<style>`` hiding the containers that belong to the other mode. 

200 

201 Containers made by :func:`advanced_only` / :func:`simple_only` stay in the 

202 page in both modes and are hidden by CSS, so switching mode never changes 

203 which elements a rerun draws, and no widget loses its value. 

204 """ 

205 advanced = parse_mode(mode) == ADVANCED 

206 hidden = f'[class*="st-key-{_SIMPLE_PREFIX if advanced else _ADV_PREFIX}"]' 

207 # In the rail, a hidden container's wrapper block would still take the 

208 # rail's gap and the margin under each block: hide the wrapper with it. A 

209 # rule of its own, so a browser without `:has()` still hides the container. 

210 hide = ( 

211 f"{hidden} {{ display: none !important; }}" 

212 ' .st-key-scanpath_rail [data-testid="stLayoutWrapper"]' 

213 f":has(> {hidden}) {{ display: none !important; }}" 

214 ) 

215 if advanced: 

216 rules = hide 

217 else: 

218 cell = '[class*="st-key-railbtn_"] > div' 

219 sort = f'> [class*="st-key-railbtn_"] > div:only-child > .st-key-{_ADV_PREFIX}{TRIAL_SORT}' 

220 rules = hide + ( 

221 # The trial summary table's wrapper would still take the page's 

222 # margin under each block. 

223 ' [data-testid="stLayoutWrapper"]' 

224 f":has(> .st-key-{_ADV_PREFIX}{TRIAL_CHIPS}) " 

225 "{ display: none !important; }" 

226 # With the slider gone its column would still hold a gap: drop it. 

227 # Only the column that holds it directly, never one around the row. 

228 ' [data-testid="stColumn"]:has(> [data-testid="stVerticalBlock"]' 

229 ' > [data-testid="stLayoutWrapper"]' 

230 f" > .st-key-{_ADV_PREFIX}{TRIAL_SLIDER})" 

231 " { display: none !important; }" 

232 # A ``railbtn_*`` cell holding only the hidden sort popover would 

233 # still take the cluster's 3px neighbour margin: drop the cell, and 

234 # the margin it gives the cell after it when it comes first. 

235 f" {cell}:has({sort}) {{ display: none !important; }}" 

236 f" {cell}:first-child:has({sort}) + div {{ margin-left: 0 !important; }}" 

237 ) 

238 return f"<style>{rules}</style>"

251def _choose_mode(mode: object) -> None: 

252 """Make ``mode`` the session's mode as the viewer's own choice. 

253 

254 Marks it chosen, which is what lets the cookie script remember it. Choosing 

255 Essentials ends a running tutorial, and a change of mode ends a running welcome 

256 tour. Anything that is not a mode (a deselected switch) changes nothing. 

257 """ 

258 import streamlit as st 

259 

260 chosen = parse_mode(mode) 

261 if chosen is not None: 

262 if chosen == SIMPLE and st.session_state.get("tutorial_active"): 

263 # The tutorials spotlight Full-only controls: leave the one 

264 # running the way its Exit does. 

265 from scanpath_studio import tour 

266 

267 tour._restore_tutorial_return() 

268 if ( 

269 chosen != current_mode() 

270 and st.session_state.get("tour_mode") == "spotlight" 

271 ): 

272 # Each mode has its own welcome tour, and a step number means a 

273 # different step in the other one: end it, as Done would. 

274 from scanpath_studio import tour 

275 

276 tour._exit_spotlight() 

277 st.session_state[UI_MODE] = chosen 

278 st.session_state[CHOSEN_KEY] = True 

279 st.session_state[COOKIE_MODE_KEY] = chosen

403def switch_css(visible: bool) -> str: 

404 """A ``<style>`` that places the Essentials | Full switch. 

405 

406 The switch stays where the script draws it, so its state never depends on 

407 the view. With ``visible`` it is lifted into Streamlit's top bar, just left 

408 of the ⋮ menu, at a larger size, and the bar's menu cell grows to leave it 

409 room. The switch never moves: Streamlit's "Running…" status widget, which 

410 the bar would draw under it, is shown just below the bar instead. Below 

411 ``_SWITCH_FALLBACK_PX`` it is an ordinary right-aligned row of the page. 

412 Without it (the views that have no use for the mode) the switch is 

413 hidden. Either way the title row it sits in takes no space of its own. 

414 """ 

415 box = f".st-key-{SWITCH_BOX_KEY}" 

416 # The cookie script's frame, which is there only once the switch has been 

417 # used, takes no room, so the switch's box (sized to its contents) and the 

418 # switch in it are the same with or without it. 

419 cookie = ( 

420 f".st-key-{COOKIE_BOX_KEY} {{ position: absolute; width: 0; height: 0;" 

421 " overflow: hidden; }" 

422 ) 

423 wide = f"@media (min-width: {_SWITCH_FALLBACK_PX + 1}px)" 

424 narrow = f"@media (max-width: {_SWITCH_FALLBACK_PX}px)" 

425 collapse = ( 

426 # Out of the flow, so the page's rows sit where they would without it; 

427 # a fixed box inside it still positions against the viewport. 

428 f"{wide} {{ {_SWITCH_ROW} {{ position: absolute; width: 0; height: 0;" 

429 " margin: 0 !important; overflow: visible; } }" 

430 ) 

431 if not visible: 

432 return ( 

433 f"<style>{box} {{ display: none !important; }} {collapse} {cookie}</style>" 

434 ) 

435 button = f'{box} [data-testid="stButtonGroup"] button[data-variant]' 

436 return ( 

437 "<style>" 

438 f"{collapse} {cookie}" 

439 f"{wide} {{" 

440 f" {box} {{ position: fixed; top: 10px; right: 3.75rem;" 

441 f" z-index: {SWITCH_Z_INDEX}; width: max-content; }}" 

442 f" {_STATUS_WIDGET_CSS}" 

443 # The header's right-hand cell is a fixed 200px: widen it so the 

444 # navigation gives way before it would run under the switch. 

445 ' [data-testid="stToolbar"] div:has(> [data-testid="stMainMenu"])' 

446 " { width: 17.5rem !important; }" 

447 " }" 

448 f" {button} {{ min-height: 2.5rem; padding: 0 1.1rem; }}" 

449 f" {button} p {{ font-size: 1rem; }}" 

450 f"{narrow} {{" 

451 # In the page: right-aligned, with the empty title column out of the way. 

452 f" {box} {{ align-items: flex-end; }}" 

453 f' [data-testid="stColumn"]:has(.st-key-about_header)' 

454 " { display: none; }" 

455 " }" 

456 "</style>" 

457 )

480def render_switch(host: DeltaGenerator, *, visible: bool = True) -> None: 

481 """Draw the Essentials | Full switch in ``host``. 

482 

483 The widget's value is re-written from the session's mode on every run, so 

484 a link or a restore that moves the mode moves the switch with it. It is 

485 drawn on every run, on every view: ``visible`` only decides, by CSS, 

486 whether it shows (see :func:`switch_css`), so nothing above a view moves 

487 with the view. Once the viewer has used it, a same-origin script keeps the 

488 choice in the browser's cookie for the next visit. 

489 """ 

490 import streamlit as st 

491 

492 from scanpath_studio.html_embed import embed_html_iframe 

493 

494 st.session_state[SWITCH_KEY] = current_mode() 

495 box = host.container(key=SWITCH_BOX_KEY) 

496 box.html(switch_css(visible)) 

497 box.segmented_control( 

498 "Interface", 

499 options=list(MODES), 

500 format_func=_LABELS.__getitem__, 

501 key=SWITCH_KEY, 

502 label_visibility="collapsed", 

503 persist_state="session", 

504 on_change=_on_switch, 

505 ) 

506 # Drawn every run, so the switch's box holds the same elements whether or 

507 # not the script is in it. 

508 cookie = box.container(key=COOKIE_BOX_KEY) 

509 if st.session_state.get(CHOSEN_KEY): 

510 with cookie: 

511 embed_html_iframe(cookie_script(cookie_mode()), height=0)
