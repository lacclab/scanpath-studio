2 

3Also serves the installed plotly package's own ``plotly.min.js`` from this app's 

4Streamlit server (ENG-64), so the in-app figures draw without a request to 

5``cdn.plot.ly`` — offline, in the desktop app, and on the hosted demo alike.

20def embed_html_iframe(html: str, *, height: int) -> None: 

21 """Render script-bearing HTML in an iframe without the deprecated API. 

22 

23 ``st.iframe`` replaced the old components embed in Streamlit 1.58. Keeping 

24 this in one module prevents the plot, guided-tour, and Share surfaces from 

25 drifting onto different embed APIs. The compatibility import is deliberately 

26 lazy so current Streamlit runs never import or call the deprecated function. 

27 """ 

28 # Tiny script-only embeds need an explicit body. Streamlit's srcdoc 

29 # autosizing observer otherwise races the parser and tries to observe a null 

30 # body, producing a browser MutationObserver error even with a fixed height. 

31 # Full Plotly documents already supply their own body. 

32 source = html 

33 if "<body" not in html.lower(): 

34 source = f"<!doctype html><html><body>{html}</body></html>" 

35 st.iframe(source, height=max(1, int(height)), tab_index=-1)

38def plotlyjs_dir() -> Path: 

39 """The installed plotly package's ``package_data`` folder. 

40 

41 It holds the ``plotly.min.js`` that plotly.py itself inlines for 

42 ``include_plotlyjs=True`` — the build its figure JSON is written for. On the 

43 frozen desktop bundle it is under the unpacked ``plotly/`` package, which the 

44 spec collects with ``collect_data_files("plotly")``. 

45 """ 

46 return Path(str(resources.files("plotly") / "package_data"))

49def plotlyjs_src() -> str: 

50 """URL of the bundled ``plotly.min.js`` on this app's own server. 

51 

52 Pass it as ``fig.to_html(include_plotlyjs=plotlyjs_src())``. Streamlit serves 

53 a registered custom-component directory at ``component/<name>/<file>`` 

54 (``declare_component(path=…)``), so the installed plotly package's own folder 

55 is registered as one — nothing is copied, and the file is always the one 

56 matching the installed plotly.py. Registration needs a script run and is 

57 idempotent, so it happens on every call rather than once at import. 

58 

59 The URL is **relative on purpose**. The figure is a ``srcdoc`` iframe, which 

60 resolves URLs against the Streamlit page's own address — the same contract 

61 Streamlit's ``index.html`` relies on for its ``./static/…`` bundle. A 

62 root-relative ``/component/…`` would miss both ``server.baseUrlPath`` and a 

63 proxy prefix such as Community Cloud's ``/~/+/``. 

64 

65 The plotly.js version is part of the component name, so an upgrade changes 

66 the URL rather than being served under a cached old one — and the URL keeps 

67 ending in ``.js``, which is what ``to_html`` needs to treat it as a script 

68 address. 

69 """ 

70 component = declare_component( 

71 f"plotlyjs-{get_plotlyjs_version()}", path=plotlyjs_dir() 

72 ) 

73 return f"component/{component.name}/{PLOTLYJS_FILENAME}"

76# Streamlit's component route answers `Cache-Control: public` with no max-age and 

77# no validator, so a browser stores the 4.8 MB script but never reuses it: every 

78# re-drawn figure (a new srcdoc document) downloaded it again, measured at the 

79# full 1.48 MB gzip body per toggle. The Streamlit page outlives its figure 

80# iframes, so the first draw leaves a Blob URL of the script on it (read back out 

81# of the HTTP cache with `force-cache`, not downloaded twice) and later draws load 

82# that. `document.write` keeps the load parser-blocking, which the figure's own 

83# inline `Plotly.newPlot` script depends on. The memo is keyed by the URL, which 

84# carries the plotly.js version, and a page that cannot be reached (not 

85# same-origin) just falls back to the plain URL. 

86_PLOTLYJS_LOADER = """<script> 

87(function () { 

88 var src = "__SRC__", url = src, host = null; 

89 try { 

90 host = window.parent !== window ? window.parent : null; 

91 if (host) void host.document; 

92 } catch (err) { 

93 host = null; 

94 } 

95 var memo = host && host.__scanpathPlotlyJs; 

96 if (memo && memo.src === src) url = memo.url; 

97 document.write('<script charset="utf-8" src="' + url + '"><\\/script>'); 

98 if (!host || url !== src) return; 

99 window.addEventListener("load", function () { 

100 fetch(src, { cache: "force-cache" }) 

101 .then(function (r) { return r.ok ? r.blob() : null; }) 

102 .then(function (blob) { 

103 var prev = host.__scanpathPlotlyJs; 

104 if (!blob || (prev && prev.src === src)) return; 

105 if (prev) host.URL.revokeObjectURL(prev.url); 

106 var copy = new host.Blob([blob], { type: "application/javascript" }); 

107 host.__scanpathPlotlyJs = { src: src, url: host.URL.createObjectURL(copy) }; 

108 }) 

109 .catch(function () {}); 

110 }); 

111})(); 

112</script>"""

115def plotlyjs_script() -> str: 

116 """The ``<script>`` that loads the bundled plotly.js into a figure iframe. 

117 

118 Put it ahead of ``fig.to_html(include_plotlyjs=False, …)``. It loads 

119 :func:`plotlyjs_src` on the first draw and, after that, the copy the 

120 Streamlit page kept of it — see ``_PLOTLYJS_LOADER`` for why. 

121 """ 

122 return _PLOTLYJS_LOADER.replace("__SRC__", plotlyjs_src())
