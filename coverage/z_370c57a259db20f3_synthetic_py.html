1"""A small, fully-specified synthetic scanpath — ground truth for testing + a 

2visualizable in-app data source ("Synthetic test trial"). 

3 

4Real eye-tracking exports are noisy and their reading measures can only be 

5checked approximately. This module ships a tiny hand-built trial whose geometry 

6and reading sequence are simple enough that *every* canonical measure has an 

7exact, hand-traced expected value (see ``EXPECTED``). The app exposes it as a 

8data source so you can eyeball the scanpath against that ground truth; the test 

9suite (``tests/test_synthetic.py``) asserts the measures match. 

10 

11Layout — 6 words, 2 lines of 3, 50x20 px boxes on a clean grid:: 

12 

13 x=100..150 x=200..250 x=300..350 

14 y=100..120 [0 The] [1 cat] [2 sat] (line 0, y-center 110) 

15 y=200..220 [3 on ] [4 the] [5 mat] (line 1, y-center 210) 

16 

17Reading sequence (9 fixations) — first-pass refixation on word 0, a regression 

18from word 2 back to word 1, and one deliberately out-of-text fixation:: 

19 

20 # t(ms) (x, y) dur lands on 

21 1 0 (115,110) 100 word 0 (first fixation) 

22 2 100 (135,110) 50 word 0 (refixation, same first pass) 

23 3 150 (225,110) 150 word 1 

24 4 300 (325,110) 120 word 2 

25 5 420 (225,110) 80 word 1 (regression back) 

26 6 500 (125,210) 200 word 3 

27 7 700 (225,210) 90 word 4 

28 8 790 (700,700) 60 -- (out of text; word_id NaN) 

29 9 850 (325,210) 110 word 5 

30""" 

31

37# Human-friendly ids so the trial reads clearly in the app's trial picker. 

38PARTICIPANT = "synthetic" 

39TRIAL = "synthetic_2line_demo" 

40PARAGRAPH = "synthetic_2line_demo"

42_WORD_TEXT = ["The", "cat", "sat", "on", "the", "mat"] 

43_WORD_X = [100, 200, 300, 100, 200, 300] 

44_WORD_Y = [100, 100, 100, 200, 200, 200] 

45_BOX_W = 50

49def make_synthetic_words() -> pd.DataFrame: 

50 """Return the normalized words/IA frame for the synthetic trial.""" 

51 n = len(_WORD_TEXT) 

52 return pd.DataFrame( 

53 { 

54 "participant_id": [PARTICIPANT] * n, 

55 "trial_id": [TRIAL] * n, 

56 "text_id": [PARAGRAPH] * n, 

57 "word_id": list(range(n)), 

58 "text": list(_WORD_TEXT), 

59 # line_idx is intentionally a constant: real OneStop IA exports do 

60 # not carry a usable per-word line number, so the by-line feature 

61 # must infer lines from geometry, not trust this column. 

62 "line_idx": [1] * n, 

63 "x": list(_WORD_X), 

64 "y": list(_WORD_Y), 

65 "width": [_BOX_W] * n, 

66 "height": [_BOX_H] * n, 

67 } 

68 ) 

69

75 (225, 110, 150, 150), 

76 (325, 110, 120, 300),

85def make_synthetic_fixations() -> pd.DataFrame: 

86 """Return the normalized fixations frame for the synthetic trial. 

87 

88 ``word_id`` is left NaN on purpose so that fixation-to-word assignment runs 

89 and can itself be tested / exercised. 

90 """ 

91 n = len(_FIX_ROWS) 

92 return pd.DataFrame( 

93 { 

94 "participant_id": [PARTICIPANT] * n, 

95 "trial_id": [TRIAL] * n, 

96 "text_id": [PARAGRAPH] * n, 

97 "x": [r[0] for r in _FIX_ROWS], 

98 "y": [r[1] for r in _FIX_ROWS], 

99 "duration_ms": [r[2] for r in _FIX_ROWS], 

100 "timestamp_ms": [r[3] for r in _FIX_ROWS], 

101 "word_id": [np.nan] * n, 

102 "order_in_trial": list(range(1, n + 1)), 

103 "pass_index": [1] * n, 

104 } 

105 )

136def make_multipart_synthetic_data() -> tuple[pd.DataFrame, pd.DataFrame]:

142 grouping, navigation, animation boundaries, and export paths have exact 

143 hand-authored expectations.
