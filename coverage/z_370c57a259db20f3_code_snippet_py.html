1"""EXP-7 — the API / CLI code that reproduces the figure currently on screen. 

2 

3The hand-off from *exploring* to *scripting*: someone tunes a figure in the app 

4for a paper, then wants that exact figure rebuilt headlessly for a batch of 

5trials without reverse-engineering which of the ~dozens of options they actually 

6changed. 

7 

8This is the **third** rendering of one state, after the deep link 

9(``url_state._build_share_query``) and the 💾 saved plot config — so it is 

10deliberately *not* a fourth reading of ``session_state``. All three take the same 

11input: the settings dict the figure was built from 

12(``controls._collect_viz_settings`` → ``tabs._build_figure_settings``). The app 

13publishes that dict, plus the trial identity and the render context, as a 

14:class:`FigureState` at the point the figure is built (``_snippet_state``), and 

15this module is a pure serializer over it with two back ends. Nothing here knows 

16about Streamlit, and nothing here decides plot semantics. 

17 

18"Only the non-defaults" is answered by :func:`api.figure_options`, which already 

19returns every figure keyword → its *effective* default. :func:`figure_kwargs` 

20diffs the live settings against it; ``explicit=True`` emits the full form 

21instead. 

22 

23The two back ends are not symmetric: the Python API takes every figure keyword 

24as itself, while the CLI spells each one as a flag in its own vocabulary. 

25:data:`_CLI_EMITTERS` is that mapping spelled out, and anything a snippet needs 

26but the CLI cannot say comes back in :attr:`ReproductionCode.cli_unsupported` — 

27a live audit of the four-surface rule rather than a silent drop. Since EXP-20 

28every figure option has a flag, so the list is empty unless an option is added 

29without one, which `tests/test_render_every_option.py` refuses. 

30"""

353def _comparable(value): 

354 """Normalize a setting for equality against its default. 

355 

356 A tuple and a list of the same numbers are the same figure — the rail 

357 produces one and the builder's signature the other — so a naive ``!=`` 

358 would report half the marker-size ranges in the app as non-default.""" 

359 if isinstance(value, (list, tuple)): 

360 return tuple(_comparable(item) for item in value) 

361 if isinstance(value, dict): 

362 return tuple(sorted((str(k), _comparable(v)) for k, v in value.items())) 

363 if isinstance(value, bool): 

364 return value 

365 if isinstance(value, (int, float)): 

366 return float(value) 

367 return value

371#: something else it already decided, so re-emitting them would be noise at best 

372#: and wrong at worst. ``connector_y`` is the extreme case: a tuple with one 

373#: float per fixation, which would bury a snippet in numbers that 

374#: ``drift_connectors=True`` recomputes anyway. 

375_DERIVED_SETTINGS = frozenset( 

376 { 

377 "connector_y", 

378 "show_connectors", 

379 "illustration_reasons",

381 # for the frame it is handed, so both snippets load that table instead 

382 # (`draws_raw_gaze`) — `show_raw_gaze=True` alone would draw nothing. 

383 "show_raw_gaze", 

384 }

388#: the trial rather than being a scalar choice. ``_DERIVED_SETTINGS`` keeps them 

389#: out of the snippet's text; this keeps them out of the published 

390#: :class:`FigureState` as well, because `tabs._amend_snippet_settings` merges

393#: never to be emitted. The rest of ``_DERIVED_SETTINGS`` stays in the state: 

394#: `show_raw_gaze` is read by :func:`state_caveats`, and the others are scalars.

407def figure_kwargs( 

408 settings: dict, kind: str = "static", *, explicit: bool = False 

409) -> dict: 

410 """The figure keywords a snippet has to pass, in ``api`` vocabulary. 

411 

412 Keys the chosen builder doesn't accept are dropped (the rail's dict carries 

413 a few, e.g. ``title_pattern``, that are not figure keywords at all), and so 

414 are keys already at their effective default — unless ``explicit``, which 

415 emits the full form. Ordered as ``figure_options`` lists them, so two 

416 snippets from neighbouring states read as neighbours.""" 

417 from . import api 

418 

419 defaults = api.figure_options(kind) 

420 out = {} 

421 for key, default in defaults.items(): 

422 if key not in settings: 

423 continue 

424 if key in _DERIVED_SETTINGS: 

425 continue 

426 value = settings[key] 

427 if key in _COMPARE_STYLE_SIDES: 

428 # The rail always builds a complete style dict, so the one an 

429 # untouched Compare carries is not a choice anyone made — only 

430 # what it changes is (EXP-20). 

431 value = compare_style_delta( 

432 value, 

433 _COMPARE_STYLE_SIDES[key], 

434 settings.get("marker_size_range", defaults.get("marker_size_range")), 

435 ) 

436 # A frame-valued option (`words_b`) is data, not a setting: its repr is 

437 # meaningless in another process. `state_caveats` names it instead. 

438 if hasattr(value, "to_dict") and hasattr(value, "columns"): 

439 continue 

440 if _is_data_uri(value): 

441 value = _IMAGE_PLACEHOLDER 

442 if explicit or _comparable(value) != _comparable(default): 

443 out[key] = value 

444 return out

451def compare_style_delta(style, idx: int, marker_size_range) -> dict | None: 

452 """What a per-scanpath style changes, or ``None`` when it changes nothing. 

453 

454 Measured against the style the comparison builder would draw *without* it 

455 (`plots._comparison_scanpath_style`), whose marker range is the figure's own 

456 ``marker_size_range`` — so a scanpath at the stock range under a changed 

457 global one still says so. Values the builder drops (``None`` / ``""`` / 

458 ``False``) are dropped here too, which is what keeps the reduced dict 

459 drawing exactly the same figure as the full one.""" 

460 if not isinstance(style, dict) or not style: 

461 return None 

462 from .plots import _comparison_scanpath_style 

463 

464 msr = tuple(marker_size_range) if marker_size_range else None 

465 kwargs = {} if msr is None else {"default_marker_size_range": msr} 

466 base = _comparison_scanpath_style(idx, None, **kwargs) 

467 drawn = _comparison_scanpath_style(idx, style, **kwargs) 

468 delta = { 

469 key: drawn[key] 

470 for key in drawn 

471 if _comparable(drawn[key]) != _comparable(base.get(key)) 

472 } 

473 return delta or None

540def _fixation_flags(value): 

541 """The PRE-2 classification dict → one ``--fixation-flag`` per category. 

542 

543 Only categories that are actually doing something are written: an *Off* 

544 category is the default, and the builder reads a missing one the same way. 

545 """ 

546 argv: list[str] = [] 

547 for category, spec in (value or {}).items(): 

548 if not isinstance(spec, dict): 

549 continue 

550 mode = str(spec.get("mode") or "Off") 

551 if mode == "Off": 

552 continue 

553 parts = [f"{category}={mode.lower()}"] 

554 # Only for the two categories that have one — `oob` / `blink` carry a 

555 # stale default threshold in the app's dict that the CLI would reject. 

556 if category in ("short", "long") and spec.get("threshold_ms") is not None: 

557 parts.append(f"threshold_ms={_num(spec['threshold_ms'])}") 

558 if spec.get("symbol"): 

559 parts.append(f"symbol={spec['symbol']}") 

560 if spec.get("color"): 

561 parts.append(f"color={spec['color']}") 

562 argv += ["--fixation-flag", ",".join(parts)] 

563 return argv

632def _matching_palette(settings: dict, kind: str) -> tuple[str | None, dict]: 

633 """The ``--palette`` a CLI snippet can name instead of spelling it out. 

634 

635 EXP-12. A palette writes eight colours at once. Spelled key by key, three of 

636 them (`text_color`, `highlight_text_color`, `background_color`) have no 

637 `render` flag and were named unsupported, and the class colours became five 

638 ``--saccade-type-color`` flags — which switch saccades to *By type*, so any 

639 palette choice produced a command drawing a different figure. A palette 

640 matches when every colour it writes that this kind draws either equals the 

641 figure's or has a flag to restate it; one that explains none (the default 

642 palette on a stock figure) is not named at all. 

643 

644 EXP-20 gave every colour a flag, which turned "can be restated" from rare 

645 into always — and exposed that a restatement is not free: a colour the 

646 figure still has at its *default* is one `figure_kwargs` never writes, so 

647 naming a palette that moves it costs a flag to move it back 

648 (`_restate_against_palette`). Before this, a figure whose text colour merely 

649 happened to equal *Print / greyscale*'s was reproduced in greyscale. The 

650 palette named is the one whose colours save the most flags net of those. 

651 

652 Returns ``(name, colours)`` — the colours the named palette supplies — or 

653 ``(None, {})``.""" 

654 from . import api 

655 from .constants import PALETTES 

656 

657 defaults = api.figure_options(kind) 

658 best: tuple[str | None, dict] = (None, {}) 

659 best_score = 0 

660 for name in PALETTES: 

661 colors = {k: v for k, v in _palette_colors(name).items() if k in defaults} 

662 score = 0 

663 for key, value in colors.items(): 

664 if key == "saccade_class_colors" and not _classes_coloured(settings): 

665 continue 

666 current = _effective_color(key, settings.get(key, defaults[key])) 

667 default = _effective_color(key, defaults[key]) 

668 at_default = _comparable(current) == _comparable(default) 

669 if _comparable(current) == _comparable(value): 

670 score += int(not at_default) 

671 elif not _flag_can_override(key, settings): 

672 break 

673 elif at_default: 

674 score -= 1 # the palette moved it; a flag has to move it back 

675 else: 

676 if score > best_score: 

677 best, best_score = (name, colors), score 

678 return best

684 """The figure keywords to write after ``--palette``: the colours it got right 

685 dropped, and every one it got wrong restated — a colour still at its default

901def _py(value) -> str: 

902 """A Python literal for a settings value. 

903 

904 ``repr`` is right for everything the settings dict holds (strings, numbers, 

905 bools, tuples, lists, dicts of those) — the one thing worth normalizing is a 

906 tuple, which reads better than a list for a fixed-arity pair.""" 

907 if isinstance(value, tuple): 

908 inner = ", ".join(_py(item) for item in value) 

909 return f"({inner})" if len(value) != 1 else f"({inner},)" 

910 if isinstance(value, list): 

911 return "[" + ", ".join(_py(item) for item in value) + "]" 

912 if isinstance(value, dict): 

913 return "{" + ", ".join(f"{_py(k)}: {_py(v)}" for k, v in value.items()) + "}" 

914 return repr(value)

923def _call_kwargs(state: FigureState, *, explicit: bool) -> list[tuple[str, Any]]: 

924 """The named (non-figure-keyword) arguments the API call carries. 

925 

926 These are parameters of ``plot_scanpath`` / ``animate_scanpath`` / 

927 ``compare_scanpaths`` rather than loose figure keywords, so they are written 

928 by hand rather than diffed — each one is emitted only when it is doing 

929 something, which is the same "only the non-defaults" rule by another route. 

930 """ 

931 out: list[tuple[str, Any]] = [] 

932 # `compare_scanpaths` has no `screen` parameter — the app pre-extracts each 

933 # side's screen before handing over the frames — so emitting one there would 

934 # be rejected as an unknown figure keyword. Reported by `state_caveats` 

935 # instead of quietly producing a snippet that raises on the first run. 

936 if state.screen and state.kind != "comparison": 

937 out.append(("screen", str(state.screen))) 

938 if state.canvas: 

939 out.append(("canvas_size", (int(state.canvas[0]), int(state.canvas[1])))) 

940 if explicit or state.base_font_size != 16: 

941 out.append(("base_font_size", int(state.base_font_size))) 

942 if state.font_family and (explicit or _non_default_font(state.font_family)): 

943 out.append(("font_family", str(state.font_family))) 

944 if state.kind == "animation": 

945 if explicit or state.playback_speed != 1.0: 

946 out.append(("playback_speed", float(state.playback_speed))) 

947 if explicit or not state.autoplay: 

948 out.append(("autoplay", bool(state.autoplay))) 

949 else: 

950 if state.fix_index_range: 

951 lo, hi = state.fix_index_range 

952 out.append(("fix_index_range", (int(lo), int(hi)))) 

953 if state.drift_correction: 

954 # The rail's own spelling is Title-case ("Warp"). `plot_scanpath` 

955 # lowercases internally but `compare_scanpaths` hands the string 

956 # straight to `alignment.correct`, so an un-lowered snippet *raises* 

957 # there. The CLI validator lowercases too — match it. 

958 out.append(("drift_correction", str(state.drift_correction).lower())) 

959 # Connectors are the static builder's alone (the original→corrected 

960 # layer has no comparison equivalent), so `compare_scanpaths` takes 

961 # no such keyword — see CLAUDE.md's render-path table. 

962 if state.drift_connectors and state.kind == "static": 

963 out.append(("drift_connectors", True))

965 # (what the app resolved it to) is in `_DERIVED_SETTINGS`, so without the

967 # with the figure on screen. `compare_scanpaths` has no such parameter — 

968 # `state_caveats` says so rather than passing a keyword it would reject. 

969 if state.kind != "comparison" and ( 

970 explicit or str(state.illustration_label).lower() != "auto" 

971 ): 

972 out.append(("illustration_label", str(state.illustration_label).lower())) 

973 if state.title: 

974 out.append(("title", str(state.title))) 

975 if state.caption: 

976 out.append(("caption", str(state.caption))) 

977 return out

1087 if explicit or compare.compare_stimulus != "both": 

1088 call.append(f" compare_stimulus={_py(compare.compare_stimulus)},") 

1089 # Emitted only when set, `explicit` included: the default is `None`, 

1090 # and the labels it stands for are composed by the builder from the 

1091 # trial ids. Writing `labels=None` would be accurate but useless, and 

1092 # writing the auto pair would freeze a derived value into the recipe. 

1093 if compare.labels: 

1094 call.append(f" labels={_py(tuple(compare.labels))},") 

1095 for name, value in _call_kwargs(state, explicit=explicit): 

1096 call.append(f" {name}={_py(value)},") 

1097 for name, value in figure_kwargs( 

1098 state.settings, state.kind, explicit=explicit 

1099 ).items(): 

1100 call.append(f" {name}={_py(value)},") 

1101 call.append(")")

1104 if output:

1106 f", {name}={_py(value)}" for name, value in (save_kwargs or {}).items() 

1107 ) 

1108 lines += ["", f"sps.save_figure(fig, {_py(output)}{extra})"]

1268def _wrap_command(argv: list[str], width: int = 76, *, env_prefix: str = "") -> str: 

1269 """One shell command, wrapped with backslash continuations. 

1270 

1271 Wrapped on flag boundaries (a token starting ``-`` opens a new group) so a 

1272 flag never ends up on a different line from its value — that is the one way 

1273 a wrapped command can be pasted and silently mean something else. 

1274 

1275 ``env_prefix`` is prepended verbatim (already shell-safe, never user text): 

1276 a flag the parser only grows under an environment variable has to be pasted 

1277 together with it.""" 

1278 groups: list[list[str]] = [] 

1279 for token in argv: 

1280 if token.startswith("-") or not groups: 

1281 groups.append([token]) 

1282 else: 

1283 groups[-1].append(token) 

1284 if env_prefix and groups: 

1285 groups[0].insert(0, env_prefix) 

1286 lines: list[str] = [] 

1287 current = "" 

1288 for group in groups: 

1289 # `env_prefix and` matters: without it an *empty* token compares equal to 

1290 # the empty default prefix and is emitted verbatim — so 

1291 # `--highlight-column ''` (mark nothing) printed as a flag with no value, 

1292 # which the parser would then read as taking the next flag as its value. 

1293 piece = " ".join( 

1294 token if (env_prefix and token == env_prefix) else shlex.quote(token) 

1295 for token in group 

1296 ) 

1297 if not current: 

1298 current = piece 

1299 elif len(current) + len(piece) + 1 <= width: 

1300 current = f"{current} {piece}" 

1301 else: 

1302 lines.append(current) 

1303 current = f" {piece}" 

1304 if current: 

1305 lines.append(current) 

1306 return " \\\n".join(lines)
