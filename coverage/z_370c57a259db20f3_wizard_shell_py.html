1"""The Add-dataset wizard's shell: part registries and status badges. 

2 

3Knows nothing about columns or dataframes — ``wizard.py`` keeps the part bodies 

4and finalize. What lives here is the *chrome*: 

5 

6- `STEPS` — the add screen's linear parts (name → data → setup, then the 

7 optional stimulus images on a local install), drawn by `part()` as numbered 

8 one-line headlines. They are labels, not navigation: there 

9 is nothing to map until a file is read, so no chips, no accordion, no open state. 

10- `EDITOR_STEPS` + `numbered()` — the ✏️ Edit dataset screen's parts, renumbered 

11 over the ones that actually render. 

12- `step_panel` — a part's heading inside the collapsed *Data & mapping* review 

13 panel. 

14"""

27class StepStatus(Enum): 

28 """What the badge beside a step says.""" 

29 

30 DONE = "done" 

31 """Satisfied — nothing more is required here.""" 

32 ACTION = "action" 

33 """Required, started, and currently blocked on something specific.""" 

34 TODO = "todo" 

35 """Required and not started.""" 

36 OPTIONAL = "optional" 

37 """Optional and untouched."""

40#: Status → its `constants.ICONS` concept. A concept rather than the 

41#: shortcode, because the badge is drawn two ways: as markdown on the review

43#: where a shortcode is inert.

52@dataclass(frozen=True) 

53class WizardStep: 

54 """One part of the wizard. ``number`` is the 1-based label the user reads.""" 

55 

56 id: str 

57 number: int 

58 title: str 

59 caption: str 

60 required: bool 

61 

62 

63#: The add screen's parts, in page order. Every table's uploader sits in its 

64#: own mapping row, with that table's *Extra fields to keep* picker directly 

65#: under its mapping, so `_part("data")` covers uploading and mapping alike. 

66#: Upload always precedes mapping (there is nothing to map until a file is 

67#: read), so the parts are labelled but not navigable: no chips, no accordion, 

68#: no open state. `part()` renders each as a one-line numbered headline. The 

69#: dataset name is its own numbered part rather than an unlabeled header above 

70#: everything, and Recording setup is its own rather than a sub-heading nested 

71#: inside "Data tables & column mapping" — they read as one flat sequence. 

72#: 

73#: The fourth, *Stimulus images* (a folder of screenshots saved with the 

74#: dataset), is optional and last. Only a local install draws it (a path on the 

75#: server means nothing to a visitor), and being last it renumbers nothing when 

76#: it is left out. 

77STEPS: tuple[WizardStep, ...] = ( 

78 WizardStep("name", 1, "Name & description", "What to call it", True), 

79 WizardStep( 

80 "data", 2, "Data tables & column mapping", "The tables you exported", True 

81 ), 

82 WizardStep("setup", 3, "Recording setup", "The screen it was recorded on", True), 

83 WizardStep( 

84 "images", 4, "Stimulus images", "Screenshots of what was on screen", False 

85 ), 

86)

91#: The ✏️ Edit dataset screen's parts, in page order. 

92#: 

93#: The two screens read the same, so the editor draws this module's `part()` 

94#: headline too. The first four ids line up one-for-one with `STEPS` — naming 

95#: it, its tables and mapping (the same question as uploading them), its 

96#: recording setup (the *same renderer*) and its stimulus images (the same 

97#: fields). The rest are the questions that only have an answer once the 

98#: dataset exists. 

99#: 

100#: The ids are prefixed ``edit_`` because `part_key` makes them container keys 

101#: and a key may be used once per run. The editor's *slots* are reserved on 

102#: every run (the screen is hidden by CSS, not skipped — see the slot comments 

103#: in `app.main`), and `edit_stimulus` is drawn on every run outside the view 

104#: guard, so an unprefixed id could meet the wizard's own while the add screen 

105#: is open. 

106#: 

107#: ``number`` here is a *placeholder*: two of the six are conditional (stimulus 

108#: images need a local filesystem, preprocessing is behind the 

109#: ``SCANPATH_EXPERIMENTAL`` flag), and a 

110#: screen numbered 1 · 2 · 3 · 5 reads as a missing section rather than as a 

111#: hidden one. `numbered()` renumbers whatever is actually on screen.

169def numbered( 

170 steps: Iterable[WizardStep], shown: Iterable[str] 

171) -> dict[str, WizardStep]: 

172 """``steps`` renumbered 1..n over the subset named in ``shown``. 

173 

174 Returns a mapping keyed by step id, so a caller can draw a part without 

175 knowing where in the sequence it landed. Ids in ``shown`` that no step 

176 declares are ignored; steps not in ``shown`` are absent from the result, 

177 which is what a caller should check rather than tracking the condition 

178 twice. 

179 """ 

180 keep = set(shown) 

181 out: dict[str, WizardStep] = {} 

182 for step in steps: 

183 if step.id not in keep: 

184 continue 

185 out[step.id] = WizardStep( 

186 step.id, len(out) + 1, step.title, step.caption, step.required 

187 ) 

188 return out

196def part( 

197 host, 

198 step: WizardStep, 

199 *, 

200 status: StepStatus | None = None, 

201 trailing=None, 

202 note: str = "", 

203): 

204 """One linear part of the wizard: a minimal headline, then its body. 

205 

206 The parts run in a fixed order — there is nothing to map before a file is 

207 read — so this is a *label*, not navigation: no expander, no chips, no open 

208 state. It costs one line, which is the point: a sequence with no choices in 

209 it needs no header and click per part. 

210 

211 ``trailing``, when given, is called with a column beside the 

212 title — a control that belongs on the title line rather than in the body — so it 

213 reads as part of the title line instead of as the first thing in the body. 

214 Only the title line splits; the returned body container stays full width. 

215 

216 ``note`` hangs the part's explanation off the title as the same hover 

217 tooltip the rest of the wizard's prose uses, instead of a `st.caption` line 

218 under it. The add screen's own titles say everything they need to, so only 

219 the ✏️ Edit dataset screen passes one — its parts carry judgements a title 

220 cannot ("does the Trial ID identify one reading?"). 

221 """ 

222 box = host.container(key=part_key(step.id)) 

223 mark = f"{icon_html(_badge_concept(status))} " if status else "" 

224 title = html.escape(step.title) 

225 if note: 

226 title = f'<span class="sps-fhelp" data-tip="{tooltip(note)}">{title}</span>' 

227 title_html = ( 

228 f'<div class="sps-wiz-part"><span class="sps-wiz-part-n">{step.number}</span>' 

229 f"{mark}{title}</div>" 

230 ) 

231 if trailing is not None: 

232 title_col, trailing_col = box.columns( 

233 [0.6, 0.4], gap="small", vertical_alignment="center" 

234 ) 

235 title_col.markdown(title_html, unsafe_allow_html=True) 

236 trailing(trailing_col) 

237 else: 

238 box.markdown(title_html, unsafe_allow_html=True) 

239 return box.container()

242def _badge_concept(status: StepStatus) -> str: 

243 return _BADGES.get(status, _BADGES[StepStatus.TODO])

246def badge(status: StepStatus) -> str: 

247 """The icon shortcode shown against a step with this status.""" 

248 return ICONS[_badge_concept(status)]

251def step_panel(host, step: WizardStep, status: StepStatus): 

252 """A part's bold heading in the collapsed *Data & mapping* review panel. 

253 

254 The panel is itself an expander, and Streamlit forbids expander-in-expander, 

255 so each part renders inline into ``host`` under this heading. The 

256 ``wizard_reconfigure`` assertion in ``tests/test_apptest.py`` relies on that. 

257 """ 

258 host.markdown(f"**{badge(status)} {step.number}. {step.title}**") 

259 return host
