1"""The ``label | field`` row — one primitive, shared by every panel that has one. 

2 

3UX-51 built this for the Scanpath rail's plot controls and UX-53 spread it over 

4the upload wizard, but both kept it private to :mod:`controls`. UX-69 needs the 

5same row in the Scanpath subtabs, and :mod:`controls` cannot supply it there: 

6it already imports :mod:`annotations` and :mod:`export`, so those two — which 

7render three of the subtabs between them — would import it back in a cycle. 

8Hence a module *below* all of them, holding nothing but the row. 

9 

10The row is two columns: the field's name on the left, the control itself on the 

11right, vertically centered against each other. That buys back the line every 

12Streamlit widget spends on its label, which is what makes a panel of a dozen 

13fields fit on one screen — and it lines the controls up on a common left edge, 

14so a column of them reads as a form rather than as a ragged stack. 

15 

16``help`` folds into the *title's* own hover tooltip rather than getting a `?` 

17icon beside it (see :func:`row_label`); the widget keeps its real label and 

18``help``, only ``label_visibility="collapsed"`` hides where Streamlit would 

19have drawn them. 

20""" 

21 

22from __future__ import annotations 

23 

24import html 

25import re 

26 

27#: The label column's share of a ``label | field`` row in a **full-width** panel 

28#: — a Scanpath subtab body, which is 4/5 of the page (UX-69). About 180px at a 

29#: typical window: enough for "Participant fields to include" to survive, while 

30#: leaving the control the room a pills row or a path pattern wants. 

31PANEL_LABEL_W = 0.2 

32 

33#: The same share in a **narrow** column: the rail's ~28rem popover (`styles. 

34#: get_app_css` pins `stPopoverBody`), or a subtab's side column. ~160px of 

35#: label — about 23 characters at the rail's 0.92rem — while leaving the field 

36#: wide enough for a multiselect's chips, or for a slider plus the UX-9 box you 

37#: type an exact value into. 

38NARROW_LABEL_W = 0.36 

39 

40#: Tighter than the 1rem default: these rows are dense and the width is scarce. 

41LABEL_GAP = "xsmall" 

42 

43#: Markdown emphasis, which a plain-text tooltip would show as literal 

44#: punctuation. 

45_MD_MARKS = re.compile(r"\*\*|`") 

46_WHITESPACE_RUN = re.compile(r"\s+") 

47

49def plain(text: str) -> str: 

50 """``text`` with markdown emphasis stripped, for a plain-text tooltip. 

51 

52 Newlines collapse to spaces along with every other run of whitespace, and 

53 that is not cosmetic (UX-68): :func:`row_label` interpolates the result into 

54 an HTML attribute inside an ``st.markdown`` string, and a **blank line** ends 

55 a raw HTML block in markdown — so a two-paragraph help text (every 

56 ``controls._gated_help`` result is one) would split the opening ``<span …>`` 

57 and render the rest of the tag as visible page text. A tooltip is one line 

58 anyway; the paragraph break has nothing to do there. 

59 """ 

60 return _WHITESPACE_RUN.sub(" ", _MD_MARKS.sub("", text)).strip()

63def row_label(host, label: str, help: str | None) -> None: 

64 """Render one row's title into its own (left) column. 

65 

66 ``help`` folds into the title's own hover tooltip rather than getting a `?` 

67 icon beside it, and the title text is repeated at the head of that tooltip so 

68 a label the column had to truncate is still readable in full. 

69 

70 The tooltip is CSS (``data-tip`` + ``styles.py``'s ``.sps-fhelp::after``), 

71 not the browser's native ``title=``. Native ``title`` waits about a second 

72 before it appears — fine for an occasional "what is this file", far too slow 

73 for a form whose every row hides its description there. The CSS one opens in 

74 ~120 ms, matching the `?` icons Streamlit draws elsewhere. ``aria-label`` 

75 keeps the text reachable now that no ``title`` carries it. 

76 """ 

77 text = plain(label) 

78 if not help: 

79 host.markdown( 

80 f'<span class="sps-flabel">{html.escape(text)}</span>', 

81 unsafe_allow_html=True, 

82 ) 

83 return 

84 tip = html.escape(f"{text} — {plain(help)}", quote=True) 

85 host.markdown( 

86 f'<span class="sps-fhelp" data-tip="{tip}" aria-label="{tip}">' 

87 f'<span class="sps-flabel sps-flabel-help">{html.escape(text)}</span>' 

88 "</span>", 

89 unsafe_allow_html=True, 

90 )

93def labeled( 

94 host, 

95 kind: str, 

96 label: str, 

97 *, 

98 display: str | None = None, 

99 help: str | None = None, 

100 label_width: float = NARROW_LABEL_W, 

101 align: str = "center", 

102 **kwargs, 

103): 

104 """Render one control as a ``label | field`` row; return the widget's value. 

105 

106 ``kind`` names the Streamlit method to call (``"selectbox"``, 

107 ``"multiselect"``, ``"color_picker"``, …) and every other argument is 

108 forwarded untouched, so converting a call site is a matter of *naming* the 

109 widget instead of calling it. The widget still receives the real ``label`` 

110 and ``help`` — only where they are drawn changes. 

111 

112 ``display`` overrides the *visible* text without touching the widget's own 

113 label. Use it where the accessible name has to stay unique but would be far 

114 too long for the column — the per-scanpath comparison styling, whose rows are 

115 already captioned with the scanpath they belong to, or a checkbox whose label 

116 is a whole sentence (the title column is one ellipsized line, so the sentence 

117 belongs in ``help``, on the hover). 

118 

119 ``label_width`` picks the column split: :data:`PANEL_LABEL_W` in a full-width 

120 panel, :data:`NARROW_LABEL_W` (the default) in the rail or a side column. 

121 ``align`` is the columns' ``vertical_alignment``; ``"top"`` suits a control 

122 that is many lines tall (a text area), where a centered title would float 

123 opposite the middle of an empty box. 

124 """ 

125 label_col, field_col = host.columns( 

126 [label_width, 1.0 - label_width], gap=LABEL_GAP, vertical_alignment=align 

127 ) 

128 row_label(label_col, display if display is not None else label, help) 

129 return getattr(field_col, kind)( 

130 label, help=help, label_visibility="collapsed", **kwargs 

131 )

134def panel_field(host, kind: str, label: str, **kwargs): 

135 """:func:`labeled` at :data:`PANEL_LABEL_W` — the full-width-panel default.""" 

136 kwargs.setdefault("label_width", PANEL_LABEL_W) 

137 return labeled(host, kind, label, **kwargs)
