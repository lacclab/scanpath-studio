58_JS = r""" 

59export default function (component) { 

60 const { data, parentElement, setTriggerValue } = component; 

61 const get = selector => parentElement.querySelector(selector); 

62 const svg = get('.authoring-canvas'); 

63 const visits = get('.visits'), duration = get('.duration'), preview = get('.preview'); 

64 const events = data.events, words = data.words; 

65 // A second click can arrive while Streamlit is applying the first. 

66 // Send the next edit when the server mounts the updated table revision. 

67 const state = svg.editorState ??= { mode: 'draw', zoom: window.matchMedia('(max-width: 600px)').matches ? '1' : 'fit', pending: [], sent: null }; 

68 clearTimeout(state.timer); 

69 const sendNext = () => { 

70 if (state.sent === null && state.pending.length && data.editable) { 

71 state.sent = data.revision; 

72 setTriggerValue('event', state.pending.shift()); 

73 } 

74 }; 

75 if (state.sent !== data.revision) 

76 state.sent = null; 

77 const emit = event => { state.pending.push(event); sendNext(); }; 

78 const make = (tag, attrs, text) => { 

79 const node = document.createElementNS('http://www.w3.org/2000/svg', tag); 

80 for (const [key, value] of Object.entries(attrs)) 

81 node.setAttribute(key, value); 

82 if (text != null) 

83 node.textContent = text; 

84 svg.appendChild(node); 

85 return node; 

86 }; 

87 const point = e => { 

88 const p = new DOMPoint(e.clientX, e.clientY).matrixTransform(svg.getScreenCTM().inverse()); 

89 return { x: Math.round(p.x * 10) / 10, y: Math.round(p.y * 10) / 10 }; 

90 }; 

91 const width = data.canvas_width, height = data.canvas_height; 

92 svg.replaceChildren(); 

93 svg.setAttribute('viewBox', `0 0 ${width} ${height}`); 

94 const zoom = get('.zoom'); 

95 const resize = () => { 

96 state.zoom = zoom.value; 

97 svg.style.width = state.zoom === 'fit' ? '100%' : `${width * Number(state.zoom)}px`; 

98 svg.style.aspectRatio = `${width} / ${height}`; 

99 }; 

100 zoom.value = state.zoom; 

101 zoom.onchange = resize; 

102 resize(); 

103 const viewport = get('.viewport'); 

104 if (data.right_to_left && !state.wasRTL) 

105 viewport.scrollLeft = viewport.scrollWidth; 

106 state.wasRTL = data.right_to_left; 

107 const mode = value => { 

108 state.mode = value; 

109 get('.draw').setAttribute('aria-pressed', value === 'draw'); 

110 get('.edit').setAttribute('aria-pressed', value === 'edit'); 

111 svg.classList.toggle('edit-mode', value === 'edit'); 

112 get('.hint').textContent = value === 'draw' ? 'Click each landing point in reading order. Click a word again to revisit it.' : 

113 'Select a fixation or drag it. Use the fixation list to reach overlapping visits.'; 

114 }; 

115 get('.draw').onclick = () => mode('draw'); 

116 get('.edit').onclick = () => mode('edit'); 

117 mode(state.mode); 

118 const labels = new Map(words.map(w => [Number(w.word_id), w.text])); 

119 for (const w of words) { 

120 make('rect', { class: 'word-box', x: w.x, y: w.y, width: w.width, height: w.height }); 

121 make('text', { class: 'word-label', x: w.x + 3, y: w.y + w.height / 2 }, w.text); 

122 } 

123 make('polyline', { class: 'path', points: events.map(e => `${e.x},${e.y}`).join(' ') }); 

124 visits.replaceChildren(new Option('Select a fixation', '')); 

125 const markers = new Map(); 

126 for (const e of events) { 

127 const id = e.fixation_id, order = e.order_in_trial; 

128 const word = labels.get(e.word_id); 

129 const description = `${order}. ${word ?? 'No target word'} · ${e.duration_ms} ms (ID ${id})`; 

130 visits.add(new Option(description, id)); 

131 const circle = make('circle', { class: 'fixation', cx: e.x, cy: e.y, r: e.diameter / 2, tabindex: 0, 

132 'aria-label': description, 'data-fixation-id': id }); 

133 const label = make('text', { class: 'order', x: e.x, y: e.y }, order); 

134 markers.set(id, { circle, label }); 

135 let start; 

136 circle.onpointerdown = ev => { 

137 if (state.mode !== 'edit' || !data.editable) 

138 return; 

139 start = point(ev); 

140 circle.setPointerCapture(ev.pointerId); 

141 }; 

142 circle.onpointermove = ev => { 

143 if (!start) 

144 return; 

145 const { x, y } = point(ev); 

146 circle.setAttribute('cx', x); 

147 circle.setAttribute('cy', y); 

148 label.setAttribute('x', x); 

149 label.setAttribute('y', y); 

150 }; 

151 circle.onpointercancel = () => { 

152 start = null; 

153 circle.setAttribute('cx', e.x); 

154 circle.setAttribute('cy', e.y); 

155 label.setAttribute('x', e.x); 

156 label.setAttribute('y', e.y); 

157 }; 

158 circle.onpointerup = ev => { 

159 if (!start) 

160 return; 

161 const pos = point(ev); 

162 emit(Math.hypot(pos.x - start.x, pos.y - start.y) > 2 

163 ? { type: 'move', fixation_id: id, ...pos } : { type: 'select', fixation_id: id }); 

164 start = null; 

165 }; 

166 circle.onkeydown = ev => { 

167 if (!data.editable) 

168 return; 

169 if (ev.key === 'Enter' || ev.key === ' ') { 

170 ev.preventDefault(); 

171 emit({ type: 'select', fixation_id: id }); 

172 } 

173 if (ev.key === 'Delete' || ev.key === 'Backspace') { 

174 ev.preventDefault(); 

175 emit({ type: 'delete', fixation_id: id }); 

176 } 

177 }; 

178 } 

179 svg.onclick = ev => { 

180 if (!data.editable || state.mode !== 'draw') 

181 return; 

182 const { x, y } = point(ev); 

183 const w = words.find(w => w.x <= x && x < w.x + w.width && w.y <= y && y < w.y + w.height); 

184 emit({ type: 'add', x, y, word_id: w?.word_id ?? null }); 

185 }; 

186 const selected = events.find(e => e.fixation_id === data.selected_fixation_id); 

187 const editable = Boolean(selected) && data.editable; 

188 visits.disabled = !events.length || !data.editable; 

189 visits.value = selected?.fixation_id ?? ''; 

190 visits.onchange = () => { if (visits.value) 

191 emit({ type: 'select', fixation_id: Number(visits.value) }); }; 

192 duration.disabled = !editable; 

193 duration.value = selected?.duration_ms ?? ''; 

194 duration.onchange = () => { 

195 if (selected && duration.value && duration.reportValidity()) 

196 emit({ type: 'update', fixation_id: selected.fixation_id, duration_ms: duration.valueAsNumber }); 

197 }; 

198 get('.delete').disabled = !editable; 

199 get('.delete').onclick = () => { if (selected) 

200 emit({ type: 'delete', fixation_id: selected.fixation_id }); }; 

201 if (selected) { 

202 const marker = markers.get(selected.fixation_id); 

203 marker.circle.classList.add('selected'); 

204 svg.append(marker.circle, marker.label); 

205 } 

206 let playing = null; 

207 svg.classList.remove('replaying'); 

208 const stop = () => { 

209 clearTimeout(state.timer); 

210 playing?.remove(); 

211 playing = null; 

212 svg.classList.remove('replaying'); 

213 preview.textContent = 'Play preview'; 

214 }; 

215 preview.textContent = 'Play preview'; 

216 preview.disabled = !events.length; 

217 preview.onclick = () => { 

218 if (playing) { 

219 stop(); 

220 return; 

221 } 

222 const play = i => { 

223 playing?.remove(); 

224 if (i === events.length) { 

225 stop(); 

226 return; 

227 } 

228 svg.classList.add('replaying'); 

229 const marker = markers.get(events[i].fixation_id); 

230 playing = make('g', { 'pointer-events': 'none', 'aria-hidden': true }); 

231 const dot = marker.circle.cloneNode(); 

232 dot.removeAttribute('tabindex'); 

233 dot.removeAttribute('data-fixation-id'); 

234 dot.setAttribute('class', 'fixation playing'); 

235 playing.append(dot, marker.label.cloneNode(true)); 

236 preview.textContent = `Stop preview (${i + 1}/${events.length})`; 

237 state.timer = setTimeout(() => play(i + 1), events[i].duration_ms); 

238 }; 

239 play(0); 

240 }; 

241 sendNext(); 

242 return () => clearTimeout(state.timer); 

243}

257def render_authoring_canvas( 

258 words: pd.DataFrame, 

259 events: pd.DataFrame, 

260 *, 

261 canvas_width: int, 

262 canvas_height: int, 

263 selected_fixation_id: int | None, 

264 revision: int, 

265 editable: bool, 

266 source: str, 

267 right_to_left: bool, 

268) -> Any: 

269 """Mount the editor and return its compact ``result.event`` trigger value.""" 

270 from .plots import _compute_marker_sizes 

271 

272 word_fields = ["word_id", "text", "line_idx", "x", "y", "width", "height"] 

273 event_fields = ["fixation_id", "order_in_trial", "x", "y", "duration_ms", "word_id"] 

274 rows = events[event_fields].copy() 

275 rows["diameter"] = ( 

276 _compute_marker_sizes(rows["duration_ms"], size_range=(20, 38)) 

277 if not rows.empty 

278 else [] 

279 ) 

280 records = rows.astype(object).where(rows.notna(), None).to_dict("records") 

281 result = _spatial_editor()( 

282 key=f"authoring_spatial_editor_{source}", 

283 data={ 

284 "words": words[word_fields].to_dict("records") if not words.empty else [], 

285 "events": records, 

286 "right_to_left": right_to_left, 

287 "revision": revision, 

288 "editable": editable, 

289 "canvas_width": canvas_width, 

290 "canvas_height": canvas_height, 

291 "selected_fixation_id": selected_fixation_id, 

292 }, 

293 on_event_change=lambda: None, 

294 height="content", 

295 ) 

296 return getattr(result, "event", None)
