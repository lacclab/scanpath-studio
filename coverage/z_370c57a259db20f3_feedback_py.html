17def _storage_answer() -> str: 

18 """Describe this session's recovery state and the relevant next action.""" 

19 local = persistence.server_bound_to_loopback() 

20 location = ( 

21 "Your files are processed on this computer." 

22 if local 

23 else "Your files are processed on this server. Use the desktop app to keep uploads on your computer." 

24 ) 

25 saving = persistence.persistence_enabled(str(st.context.url or "")) 

26 backups = ( 

27 "Keep your original files. Before refreshing, export annotations from " 

28 "**Data Management → Current dataset → Annotations** and plot settings from " 

29 "**Scanpath → Share → Settings file**." 

30 ) 

31 if not saving: 

32 recovery = f"Uploads stay in memory for this session. {backups}" 

33 elif st.session_state.get(persistence.SAVE_CONFLICT_KEY) or st.session_state.get( 

34 persistence.SAVE_FAILED_KEY 

35 ): 

36 recovery = f"The app couldn’t update the recovery copy. Your unsaved edits are still in this tab. {backups}" 

37 elif persistence.persistence_paused(st.session_state): 

38 recovery = f"Recovery saving is paused for this session. {backups} Reload to retry the saved copy." 

39 else: 

40 recovery = ( 

41 "A recovery copy keeps your datasets, settings and annotations between visits. " 

42 "In **Data Management → Saved work**, open **Your session → " 

43 "Saved session details → Clear saved session…** to remove it." 

44 ) 

45 return f"{location} {recovery}"

133def maybe_show_feedback() -> None: 

134 """Open the requested dialog before the app draws the current plot.""" 

135 if st.session_state.pop(_REQUESTED_KEY, False): 

136 _feedback_dialog()

139@st.dialog(f"{ICONS['feedback']} Troubleshooting & feedback", width="medium") 

140@guarded() 

141def _feedback_dialog() -> None: 

142 for problem, answer in troubleshooting_items(): 

143 with st.expander(problem): 

144 st.markdown(answer) 

145 

146 st.markdown("**Contact the project on GitHub**") 

147 with st.container(horizontal=True, gap="small"): 

148 st.link_button("Ask a question ↗", CITATION["questions_url"]) 

149 reporting = st.button("Report a bug", key="help_report_bug") 

150 st.link_button("Suggest an improvement ↗", CITATION["ideas_url"]) 

151 if reporting: 

152 st.markdown( 

153 "Tell us what you did, what you expected and what happened. " 

154 "Copy these details into the GitHub report." 

155 ) 

156 st.code(environment_details(), language=None, wrap_lines=True) 

157 st.link_button("Open bug report on GitHub ↗", CITATION["bug_report_url"])
