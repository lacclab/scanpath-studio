1"""How to cite Scanpath Studio: the BibTeX and APA entries. 

2 

3One builder for every place the citation is shown — ❓ Help → About in the app 

4and the docs site's *Cite* section (``scripts/docs_support.py``) — so the two 

5print the same entry. It reads ``constants.CITATION`` / ``CITATION_AUTHORS`` and 

6the package's ``__release__``, which ``tests/test_citation.py`` keeps equal to 

7``CITATION.cff`` (the file behind GitHub's *Cite this repository* and the Zenodo 

8record); the installed package has no ``CITATION.cff`` to read. 

9""" 

10 

11from __future__ import annotations 

12 

13import unicodedata 

14 

15from scanpath_studio.constants import CITATION, CITATION_AUTHORS 

16 

17#: Combining mark → the BibTeX accent command that spells it, so the entry 

18#: compiles under classic BibTeX as well as biber. 

19_BIBTEX_ACCENTS = { 

20 "̀": "`", # grave 

21 "́": "'", # acute 

22 "̂": "^", # circumflex 

23 "̃": "~", # tilde 

24 "̈": '"', # diaeresis 

25 "̊": "r", # ring 

26 "̌": "v", # caron 

27 "̧": "c", # cedilla 

28} 

29 

30 

31def bibtex_text(text: str) -> str: 

32 """``Jäger`` → ``J{\\"a}ger``: every accented letter as an escaped group.""" 

33 out: list[str] = [] 

34 for char in unicodedata.normalize("NFD", text): 

35 accent = _BIBTEX_ACCENTS.get(char) 

36 if accent and out: 

37 base = out.pop() 

38 sep = " " if accent.isalpha() else "" 

39 out.append(f"{{\\{accent}{sep}{base}}}") 

40 else: 

41 out.append(char) 

42 return unicodedata.normalize("NFC", "".join(out)) 

43 

44 

45def _initials(given: str) -> str: 

46 """``Deborah N.`` → ``D. N.``; ``Jean-Paul`` → ``J.-P.`` (APA 7).""" 

47 parts = [] 

48 for name in given.split(): 

49 if name.endswith(".") and len(name) <= 3: 

50 parts.append(name) 

51 else: 

52 parts.append("-".join(f"{piece[0]}." for piece in name.split("-"))) 

53 return " ".join(parts) 

54 

55 

56def _release() -> str: 

57 from scanpath_studio import __release__ 

58 

59 return __release__ 

60 

61 

62def software_bibtex() -> str: 

63 """The software's BibTeX entry, citing the Zenodo concept DOI.""" 

64 year = CITATION["year"] 

65 doi = CITATION["doi"] 

66 # One author per line, so the entry fits a code block without scrolling. 

67 authors = " and\n ".join( 

68 f"{bibtex_text(family)}, {bibtex_text(given)}" 

69 for given, family in CITATION_AUTHORS 

70 ) 

71 return "\n".join( 

72 [ 

73 f"@software{{Shubi_Scanpath_Studio_{year},", 

74 f" author = {{{authors}}},", 

75 f" title = {{{{{CITATION['title']}}}}},", 

76 f" year = {{{year}}},", 

77 f" version = {{{_release()}}},", 

78 f" doi = {{{doi}}},", 

79 f" url = {{https://doi.org/{doi}}},", 

80 f" license = {{{CITATION['license']}}},", 

81 "}", 

82 ] 

83 ) 

84 

85 

86def software_apa() -> str: 

87 """The software's APA 7 reference, in Markdown (the title in italics).""" 

88 names = [f"{family}, {_initials(given)}" for given, family in CITATION_AUTHORS] 

89 listed = ", ".join(names[:-1]) + f", & {names[-1]}" if len(names) > 1 else names[0] 

90 return ( 

91 f"{listed} ({CITATION['year']}). *{CITATION['title']}* " 

92 f"(Version {_release()}) [Computer software]. " 

93 f"https://doi.org/{CITATION['doi']}" 

94 )
