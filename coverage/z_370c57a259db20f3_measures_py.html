1"""Compute reading measures from fixations and word bounding boxes. 

2 

3Definitions follow standard reading-research conventions (Rayner 1998; 

4Inhoff & Radach 1998). All measures are per (participant, trial, word). When a 

5column already exists on the words dataframe (e.g. pre-aggregated EyeLink IA 

6metrics) it is preserved; we only compute values that are not present. 

7 

8Canonical output columns added to words: 

9- first_fixation_ms — FFD: duration of the first fixation on this word 

10- first_pass_gaze_ms — FPRT / gaze duration 

11- regression_path_ms — RPD / go-past time 

12- total_fixation_ms — TFD / dwell 

13- n_fixations — fixation count 

14- skip_flag — True if no first-pass fixation 

15- regression_in_flag — True if any later fixation returned here 

16- regression_out_flag — True if a fixation here was followed by a regression 

17- first_fix_x, first_fix_y — landing position of the first-pass first fixation 

18 

19The fixations dataframe is also enriched with: 

20- word_id — assigned via bbox containment + nearest-word fallback 

21- saccade_amplitude — pixel distance from the previous fixation in the trial. 

22 Always pixels (BUG-25): EyeLink's degree-valued 

23 NEXT_SAC_AMPLITUDE / PREVIOUS_SAC_AMPLITUDE keep their 

24 own `*_deg` names rather than sharing this one. 

25- progression — 1 if the next fixation moves to a later word, -1 if earlier, 0 otherwise 

26- is_regression — True if this fixation lands on a word earlier than the 

27 running maximum word reached in the trial 

28"""

37# Default line-misregistration tolerance (px): a fixation that falls outside 

38# every word box snaps to the nearest word centre within this radius before it 

39# is left unassigned. Shared by the grouped assigner here and the single-frame 

40# helper used for model scanpaths in :mod:`scanpath_studio.similarity`. 

41LINE_MISREGISTRATION_PX = 50.0 

42 

43# A recorded ``timestamp_ms`` series is trusted as real reading time only when 

44# its span covers at least this fraction of the summed fixation durations. 

45# Fixations don't overlap, so a genuine recording spans at least its total 

46# dwell; the 0,1,2,… row index ``data.normalize_fixations`` synthesises when the 

47# source has no timestamps collapses to a few ms and must NOT be read as 

48# milliseconds. Shared by the similarity time-curve and the animation clock. 

49REAL_TIMESTAMP_DWELL_FRAC = 0.5

52def _assign_word_ids_single( 

53 fix_chunk: pd.DataFrame, 

54 word_chunk: pd.DataFrame, 

55 nearest_within_px: float = LINE_MISREGISTRATION_PX, 

56) -> np.ndarray: 

57 """Vectorized fixation→word_id assignment for a single trial's frames. 

58 

59 Tests every fixation in ``fix_chunk`` against every word box in 

60 ``word_chunk`` (no participant/trial grouping — the caller is responsible 

61 for slicing to one trial, or for passing frames whose ids deliberately 

62 don't match, as the model scanpaths do). Fixations outside every box snap 

63 to the nearest word centre within ``nearest_within_px`` (line- 

64 misregistration tolerance), else NaN. 

65 

66 Returns a float array aligned to ``fix_chunk`` rows (NaN = out of text). 

67 """ 

68 wx0, wy0, wx1, wy1 = word_box_bounds(word_chunk) 

69 wids = word_chunk["word_id"].to_numpy() 

70 wcx = (wx0 + wx1) / 2.0 

71 wcy = (wy0 + wy1) / 2.0 

72 

73 fx = pd.to_numeric(fix_chunk["x"], errors="coerce").to_numpy(dtype=float) 

74 fy = pd.to_numeric(fix_chunk["y"], errors="coerce").to_numpy(dtype=float) 

75 

76 in_box = ( 

77 (fx[:, None] >= wx0[None, :]) 

78 & (fx[:, None] <= wx1[None, :]) 

79 & (fy[:, None] >= wy0[None, :]) 

80 & (fy[:, None] <= wy1[None, :]) 

81 ) 

82 word_idx = np.where(in_box.any(axis=1), in_box.argmax(axis=1), -1) 

83 

84 # Fallback: nearest word center within nearest_within_px. 

85 unassigned = word_idx == -1 

86 if unassigned.any() and nearest_within_px > 0: 

87 dists = np.sqrt( 

88 (fx[unassigned, None] - wcx[None, :]) ** 2 

89 + (fy[unassigned, None] - wcy[None, :]) ** 2 

90 ) 

91 nearest = dists.argmin(axis=1) 

92 within = dists[np.arange(len(nearest)), nearest] <= nearest_within_px 

93 word_idx[unassigned] = np.where(within, nearest, -1) 

94 

95 return np.where(word_idx >= 0, wids[np.clip(word_idx, 0, None)], np.nan)

98# --- BUG-11 · word-box boundaries that sit mid-space ------------------------ 

99# An interest-area boundary should fall in the MIDDLE of the whitespace between 

100# two words. In some corpora it doesn't. Interest areas are defined by the 

101# *experiment*, not by the tracker — EyeLink Data Viewer just measures against 

102# whatever rectangles the IAS file gives it — and a stimulus generator that 

103# tiles a monospaced line hands every box the whole following inter-word space 

104# as trailing padding. The boxes then tile with no gaps, but every boundary sits 

105# a half-space too far right, and a fixation landing in the space *before* a 

106# word is credited to the previous word. On the bundled demo every box is 

107# exactly ``(n_chars + 1) × advance`` px wide and starts at the word's first 

108# glyph (checked against the corpus's own stimulus images: per line the ink 

109# starts within 3 px of the first box's left edge and stops ~1 advance short of 

110# the last box's right edge). 

111# 

112# The correction has to be conditional: glyph-tight AOIs (PoTeC, MultiplEYE) 

113# leave real gaps between boxes and are already centred by construction, so 

114# shifting them would introduce the very error this fixes. `word_box_space_px` 

115# therefore only reports a padding width when the layout actually looks like 

116# "tiling boxes with one trailing space", and returns 0.0 otherwise. 

117# 

118# Only the BOUNDARY moves. ``x`` keeps meaning "where the glyphs start", so the 

119# true-to-scale word labels and the stimulus-image alignment (BUG-3 / VIZ-4) are 

120# untouched — the labels are drawn from the original frame, the boundaries from 

121# the recentred one. 

122_TILING_GAP_TOL_PX = 1.0 

123# Box widths are integers in the exports, so `(n_chars + 1) × advance` is only 

124# met to within a pixel of rounding; and a stray token (a stripped-out glyph, a 

125# joined punctuation mark) shouldn't disqualify an otherwise regular layout. 

126_ADVANCE_RESIDUAL_TOL_PX = 1.5 

127_ADVANCE_MIN_AGREEMENT = 0.95

130def _sample_trial_words(words: pd.DataFrame) -> pd.DataFrame: 

131 """One trial's words — the layout is a property of the corpus, not a trial. 

132 

133 Necessary as well as cheap: every trial is drawn at the same screen 

134 positions, so clustering lines across a whole corpus would merge different 

135 trials' words into one "line" and read their interleaved x as gaps. 

136 """ 

137 keys = grouping_columns(words) 

138 if not keys: 

139 return words 

140 first = words[keys].iloc[0] 

141 mask = pd.Series(True, index=words.index) 

142 for key in keys: 

143 mask &= words[key] == first[key] 

144 return words[mask]

147def word_box_space_px(words: pd.DataFrame) -> float: 

148 """Width of the trailing inter-word padding baked into each box, or ``0.0``. 

149 

150 Non-zero only for a monospaced, *tiling* layout whose boxes are consistently 

151 ``(n_chars + 1)`` advances wide — the shape that carries the whole space as 

152 trailing padding. Anything else (glyph-tight AOIs, proportional fonts, 

153 missing text) reports 0.0, i.e. "don't touch this layout". 

154 """ 

155 needed = {"x", "width", "text"} 

156 if words is None or words.empty or not needed <= set(words.columns): 

157 return 0.0 

158 sample = _sample_trial_words(words) 

159 x = pd.to_numeric(sample["x"], errors="coerce") 

160 width = pd.to_numeric(sample["width"], errors="coerce") 

161 chars = sample["text"].astype(str).str.len() 

162 ok = x.notna() & width.notna() & (chars > 0) & (width > 0) 

163 if ok.sum() < 3: 

164 return 0.0 

165 # One advance per character plus exactly one for the trailing space. 

166 advance = float((width[ok] / (chars[ok] + 1)).median()) 

167 if advance <= 0: 

168 return 0.0 

169 residual = (width[ok] - (chars[ok] + 1) * advance).abs() 

170 if (residual <= _ADVANCE_RESIDUAL_TOL_PX).mean() < _ADVANCE_MIN_AGREEMENT: 

171 return 0.0 # not monospaced-with-one-trailing-space 

172 # And the boxes must actually tile: a layout with real gaps is glyph-tight, 

173 # so its boundaries already sit in the whitespace. 

174 lines = cluster_word_lines(sample) 

175 for _, line_words in sample[ok].groupby(lines[ok], sort=False): 

176 ordered = line_words.sort_values("x") 

177 if len(ordered) < 2: 

178 continue 

179 left = pd.to_numeric(ordered["x"], errors="coerce").to_numpy() 

180 right = left + pd.to_numeric(ordered["width"], errors="coerce").to_numpy() 

181 gaps = left[1:] - right[:-1] 

182 if len(gaps) and np.nanmax(np.abs(gaps)) > _TILING_GAP_TOL_PX: 

183 return 0.0 

184 return advance

187def word_box_bounds( 

188 words: pd.DataFrame, 

189 *, 

190 layout: pd.DataFrame | None = None, 

191) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]: 

192 """``(x0, y0, x1, y1)`` AOI edges with the mid-space correction applied. 

193 

194 **The** accessor for word-box geometry: anything that tests a point against a 

195 box, draws one, or measures a position within one goes through here, so the 

196 boundary is in one place instead of re-derived per call site. Pure — it reads 

197 ``x``/``y``/``width``/``height`` and returns arrays, never a shifted frame, so 

198 it cannot be applied twice by accident (the failure mode that made the first 

199 pass at BUG-11 fragile). For a glyph-tight corpus the correction is zero and 

200 these are the raw edges. 

201 

202 Pass ``layout`` when ``words`` is a *subset* of a trial (a highlighted span, 

203 the words that got any dwell): tiling is a property of the whole line, and a 

204 subset has holes in it, so detection has to run on the full frame or it reads 

205 the holes as glyph-tight gaps and silently declines to correct. 

206 """ 

207 if words is None or words.empty: 

208 # A column-less empty frame is a legitimate "no words" input. 

209 empty = np.empty(0, dtype=float) 

210 return empty, empty.copy(), empty.copy(), empty.copy() 

211 x = pd.to_numeric(words["x"], errors="coerce").to_numpy(dtype=float) 

212 y = pd.to_numeric(words["y"], errors="coerce").to_numpy(dtype=float) 

213 w = pd.to_numeric(words["width"], errors="coerce").to_numpy(dtype=float) 

214 h = pd.to_numeric(words["height"], errors="coerce").to_numpy(dtype=float) 

215 x0 = x - word_box_space_px(words if layout is None else layout) / 2.0 

216 return x0, y, x0 + w, y + h

219def word_char_advance( 

220 words: pd.DataFrame, 

221 *, 

222 layout: pd.DataFrame | None = None, 

223 chars: np.ndarray | None = None, 

224) -> np.ndarray: 

225 """Width of one character in each word, in px — the *within-word* scale. 

226 

227 **The** accessor for anything that measures a position *inside* a word in 

228 letters (initial landing position, a saccade's launch/landing letter), the 

229 way :func:`word_box_bounds` is the accessor for the boundary between words. 

230 

231 VAL-5 found the two disagreeing. The boundary had been corrected for BUG-11's 

232 trailing inter-word padding while the letter scale had not: ``width / 

233 len(text)`` divides a box that is ``len(text) + 1`` advances wide by 

234 ``len(text)``, so on a tiling corpus every letter was reported ~``(n+1)/n`` 

235 too wide and every landing position that far into the word. The fix is the 

236 same detection, applied to the denominator — ``width / (len(text) + 1)`` for 

237 a tiling layout, ``width / len(text)`` for a glyph-tight one. 

238 

239 The *origin* of a within-word position stays the word's ``x`` (where the 

240 glyphs start), not the corrected boundary: half an inter-word space to the 

241 left of the first glyph is where the AOI begins, not where the word does. 

242 With the advance fixed, ``x`` and the corrected boundary are one advance/2 

243 apart by construction, so the two accessors describe one consistent geometry. 

244 

245 ``layout`` has the same meaning as in :func:`word_box_bounds`: pass the full 

246 trial when ``words`` is a subset, since tiling is a property of the line. 

247 

248 ``chars`` lets a caller that already counted the characters hand them in — 

249 ``.astype(str).str.len()`` is a Python-level pass over the frame, and 

250 ``aggregation._glyph_span`` needs the same counts for the glyph run. 

251 """ 

252 if words is None or words.empty or not {"width", "text"} <= set(words.columns): 

253 # NaN, never `np.empty` — that returns *uninitialized* floats, and a 

254 # garbage positive one passes the `isfinite and > 0` guard at the call 

255 # site and lands a silently wrong landing position in a cached frame. 

256 return np.full(0 if words is None else len(words), np.nan, dtype=float) 

257 width = pd.to_numeric(words["width"], errors="coerce").to_numpy(dtype=float) 

258 counts = word_char_counts(words) if chars is None else chars 

259 padded = word_box_space_px(words if layout is None else layout) > 0 

260 return width / (counts + (1.0 if padded else 0.0))

268def recentre_word_boxes(words: pd.DataFrame) -> pd.DataFrame: 

269 """The :func:`word_box_bounds` correction as a frame, for row-wise consumers. 

270 

271 Returns ``words`` unchanged (the same object) for layouts 

272 :func:`word_box_space_px` doesn't recognise, so the common case costs one 

273 check and no copy. Prefer :func:`word_box_bounds` — this shifts ``x``, so a 

274 frame that has been through it must not go through it again. 

275 """ 

276 space = word_box_space_px(words) 

277 if space <= 0: 

278 return words 

279 out = words.copy() 

280 out["x"] = pd.to_numeric(out["x"], errors="coerce") - space / 2.0 

281 return out

284def assign_fixations_to_words( 

285 fixations: pd.DataFrame, 

286 words: pd.DataFrame, 

287 *, 

288 overwrite: bool = False, 

289 nearest_within_px: float = LINE_MISREGISTRATION_PX, 

290) -> pd.DataFrame: 

291 """Assign each fixation to a word via bounding-box containment. 

292 

293 If a fixation does not fall inside any word box, assign it to the nearest 

294 word center within `nearest_within_px` pixels (a common practice for line 

295 misregistration). Beyond that radius, the fixation gets word_id=NaN. 

296 

297 If `overwrite=False` and the fixations already carry word_id values, those 

298 are kept; only NaN rows get re-assigned. 

299 

300 Box edges come from :func:`word_box_bounds` (BUG-11), so a fixation in the 

301 whitespace *before* a word is credited to that word rather than the previous 

302 one. 

303 """ 

304 if fixations.empty or words.empty: 

305 return fixations 

306 

307 out = fixations.copy() 

308 if "word_id" not in out.columns or overwrite: 

309 out["word_id"] = np.nan 

310 

311 need_idx = out["word_id"].isna() 

312 if not need_idx.any(): 

313 return out 

314 

315 # Per (participant, trial), do a fast vectorized box-test against that 

316 # trial's words. 

317 keys = grouping_columns(out) 

318 if keys != grouping_columns(words): 

319 raise ValueError("Words and fixations use different multipart identities.") 

320 groups = out[need_idx].groupby(keys, sort=False) 

321 word_groups = words.groupby(keys, sort=False) 

322 

323 assignments = pd.Series(np.nan, index=out.index[need_idx], dtype=float) 

324 for group_key, fix_chunk in groups: 

325 lookup_key = group_key if len(keys) > 1 else group_key[0] 

326 try: 

327 wchunk = word_groups.get_group(lookup_key) 

328 except KeyError: 

329 continue 

330 if wchunk.empty: 

331 continue 

332 assignments.loc[fix_chunk.index] = _assign_word_ids_single( 

333 fix_chunk, wchunk, nearest_within_px 

334 ) 

335 

336 out.loc[need_idx, "word_id"] = assignments 

337 return out

340def materialize_runs(fixations: pd.DataFrame) -> pd.DataFrame: 

341 """Add trial run, line run, and per-word visit/pass columns (PRE-16).""" 

342 if fixations.empty: 

343 return fixations.copy() 

344 derived = ( 

345 "run", 

346 "linerun", 

347 "word_runid", 

348 "word_run", 

349 "word_run_fix", 

350 "nrun", 

351 "reread", 

352 ) 

353 out = fixations.drop(columns=[c for c in derived if c in fixations]).copy() 

354 if "word_id" not in out: 

355 out["word_id"] = pd.Series(pd.NA, index=out.index, dtype="Float64") 

356 excluded = out.get("excluded", pd.Series(False, index=out.index)).fillna(False) 

357 active = out.loc[~excluded.astype(bool)].copy() 

358 for column in derived: 

359 out[column] = pd.NA 

360 if active.empty: 

361 return out.sort_index() 

362 keys = grouping_columns(active) 

363 order_cols = keys + (["timestamp_ms"] if "timestamp_ms" in out else []) 

364 active = active.sort_values(order_cols, kind="stable") 

365 group = active.groupby(keys, sort=False, dropna=False) if keys else [((), active)] 

366 pieces = [] 

367 for _, chunk in group: 

368 chunk = chunk.copy() 

369 word = pd.to_numeric( 

370 chunk.get("word_id", pd.Series(pd.NA, index=chunk.index)), 

371 errors="coerce", 

372 ) 

373 line_source = ( 

374 chunk["line_id"] 

375 if "line_id" in chunk 

376 else chunk.get("line_idx", pd.Series(pd.NA, index=chunk.index)) 

377 ) 

378 line = pd.to_numeric(line_source, errors="coerce") 

379 chunk["run"] = (word.diff().fillna(0) < 0).cumsum().astype("Int64") + 1 

380 chunk["linerun"] = ( 

381 (line.ne(line.shift()) & line.notna()).cumsum().astype("Int64") 

382 ) 

383 visit_start = word.ne(word.shift()) | word.isna() 

384 chunk["word_runid"] = visit_start.cumsum().astype("Int64") 

385 valid = chunk[word.notna()].copy() 

386 if not valid.empty: 

387 visits = valid[["word_id", "word_runid"]].drop_duplicates() 

388 visits["word_run"] = visits.groupby("word_id", sort=False).cumcount() + 1 

389 lookup = visits.set_index(["word_id", "word_runid"])["word_run"] 

390 visit_keys = pd.MultiIndex.from_frame(chunk[["word_id", "word_runid"]]) 

391 chunk["word_run"] = lookup.reindex(visit_keys).to_numpy() 

392 chunk["word_run_fix"] = ( 

393 chunk.groupby(["word_id", "word_runid"], dropna=False).cumcount() + 1 

394 ) 

395 else: 

396 chunk["word_run"] = pd.NA 

397 chunk["word_run_fix"] = pd.NA 

398 chunk["nrun"] = chunk.groupby("word_id", dropna=False)["word_runid"].transform( 

399 "nunique" 

400 ) 

401 chunk["reread"] = pd.to_numeric(chunk["word_run"], errors="coerce").gt(1) 

402 pieces.append(chunk) 

403 computed = pd.concat(pieces).sort_index() if pieces else active 

404 for column in derived: 

405 out.loc[computed.index, column] = computed[column] 

406 return out.sort_index()

409def enrich_fixations(fixations: pd.DataFrame, words: pd.DataFrame) -> pd.DataFrame: 

410 """Add saccade_amplitude, progression, and is_regression to fixations.""" 

411 if fixations.empty: 

412 return fixations 

413 out = fixations.copy() 

414 keys = grouping_columns(out) 

415 out = out.sort_values(keys + ["timestamp_ms"]) 

416 

417 g = out.groupby(keys, sort=False) 

418 dx = g["x"].diff() 

419 dy = g["y"].diff() 

420 if "saccade_amplitude" not in out.columns: 

421 out["saccade_amplitude"] = np.sqrt(dx * dx + dy * dy) 

422 else: 

423 out["saccade_amplitude"] = out["saccade_amplitude"].where( 

424 out["saccade_amplitude"].notna(), np.sqrt(dx * dx + dy * dy) 

425 ) 

426 incoming = np.degrees(np.arctan2(-dy, dx)) 

427 out["angle_incoming"] = incoming 

428 g = out.groupby(keys, sort=False) 

429 out["angle_outgoing"] = g["angle_incoming"].shift(-1) 

430 

431 next_word = g["word_id"].shift(-1) 

432 out["progression"] = np.sign(next_word - out["word_id"]).fillna(0).astype(int) 

433 

434 running_max = g["word_id"].cummax() 

435 out["is_regression"] = (out["word_id"] < running_max).fillna(False).astype(bool) 

436 return materialize_runs(out)

439def classify_saccades(fixations: pd.DataFrame, words: pd.DataFrame) -> pd.Series: 

440 """Classify each *outgoing* saccade by its reading type (VIZ-8). 

441 

442 A saccade is the segment from one fixation to the next in reading order; its 

443 class is stored on the *departing* fixation, so the last fixation of every 

444 trial (which has no outgoing saccade) is ``None``. Classes follow the classic 

445 reading schematic: 

446 

447 - ``refixation`` — lands back on the same word (``word_id`` unchanged), 

448 - ``regression`` — moves to an earlier line, or backward within a line, 

449 - ``return_sweep`` — sweeps down to a later line (forward reading), 

450 - ``forward`` — advances to the next word on the same line, 

451 - ``skip`` — advances past one or more words on the same line, 

452 - ``other`` — a real saccade whose endpoints can't be classified (an 

453 out-of-text fixation with no assigned word). 

454 

455 Line membership comes from :func:`assign_fixation_lines` (word-box geometry), 

456 word order from the fixation ``word_id`` assignment. Returns an object Series 

457 aligned to ``fixations.index`` (pure — no plotting), so both the render path 

458 (``plots._add_saccade_layer``) and any future analysis can reuse it. Works on 

459 a single already-sliced trial or a multi-trial frame (grouped by 

460 participant/trial). 

461 """ 

462 if fixations.empty: 

463 return pd.Series([], dtype=object, index=fixations.index) 

464 

465 keys = grouping_columns(fixations) 

466 sort_cols = keys + (["timestamp_ms"] if "timestamp_ms" in fixations.columns else []) 

467 order = fixations.sort_values(sort_cols).index if sort_cols else fixations.index 

468 

469 if "word_id" in fixations.columns: 

470 word = pd.to_numeric(fixations["word_id"], errors="coerce").reindex(order) 

471 else: 

472 word = pd.Series(np.nan, index=order, dtype=float) 

473 if words is not None and not words.empty: 

474 line = assign_fixation_lines(fixations, words).reindex(order) 

475 else: 

476 line = pd.Series(np.nan, index=order, dtype=float) 

477 

478 if keys: 

479 grp = [fixations[k].reindex(order) for k in keys] 

480 next_word = word.groupby(grp).shift(-1) 

481 next_line = line.groupby(grp).shift(-1) 

482 next_pos = pd.Series(np.arange(len(order)), index=order).groupby(grp).shift(-1) 

483 else: 

484 next_word = word.shift(-1) 

485 next_line = line.shift(-1) 

486 next_pos = pd.Series(np.arange(len(order)), index=order).shift(-1) 

487 

488 word_arr = word.to_numpy(dtype=float) 

489 next_word_arr = next_word.to_numpy(dtype=float) 

490 dw = next_word_arr - word_arr 

491 dl = (next_line - line).to_numpy(dtype=float) 

492 has_next = next_pos.notna().to_numpy() 

493 # A saccade is classifiable only when BOTH endpoints have an assigned word; 

494 # otherwise the word/line deltas are meaningless (an off-text fixation still 

495 # gets a *nearest* line, which would spuriously read as a line change). 

496 both_words = ~np.isnan(word_arr) & ~np.isnan(next_word_arr) 

497 same_line = np.isnan(dl) | (dl == 0) 

498 

499 # Priority order: same word first, then line moves, then within-line word moves. 

500 conds = [ 

501 dw == 0, # refixation 

502 dl < 0, # regression — up to an earlier line 

503 dl > 0, # return sweep — down to a later line 

504 same_line & (dw < 0), # regression — backward within the line 

505 same_line & (dw == 1), # forward 

506 same_line & (dw >= 2), # skip 

507 ] 

508 choices = [ 

509 "refixation", 

510 "regression", 

511 "return_sweep", 

512 "regression", 

513 "forward", 

514 "skip", 

515 ] 

516 classed = np.select(conds, choices, default="other").astype(object) 

517 classed[~both_words] = "other" # a real saccade, but an endpoint is off-text 

518 classed[~has_next] = None # last fixation of a trial: no outgoing saccade 

519 # dtype=object is load-bearing: pandas 3.0 infers a dedicated `str` dtype for 

520 # an all-string+None array and coerces the None sentinel to NaN, breaking the 

521 # "None for the last fixation" contract. Pin object so None survives. 

522 return pd.Series(classed, index=order, dtype=object).reindex(fixations.index)

525def rebased_fixation_onsets(ordered_fixations: pd.DataFrame) -> np.ndarray: 

526 """Fixation onset times (ms), rebased so the first fixation is t=0. 

527 

528 ``ordered_fixations`` must already be in reading order (sorted by 

529 ``timestamp_ms``); the returned array is aligned to its rows. Uses the 

530 recorded ``timestamp_ms`` when they look like real times — their span is at 

531 least ``REAL_TIMESTAMP_DWELL_FRAC`` of the summed durations — otherwise lays 

532 fixations back-to-back by their durations, so a synthesised 0,1,2,… index 

533 doesn't crush the time axis. Shared by the similarity time-curve 

534 (:func:`scanpath_studio.similarity._rebased_onsets`) and the animation clock 

535 (:func:`scanpath_studio.plots._scanpath_anim_specs`). 

536 """ 

537 if ordered_fixations.empty: 

538 return np.array([], dtype=float) 

539 if "duration_ms" in ordered_fixations.columns: 

540 dur = ( 

541 pd.to_numeric(ordered_fixations["duration_ms"], errors="coerce") 

542 .fillna(0) 

543 .to_numpy(dtype=float) 

544 ) 

545 else: 

546 dur = np.zeros(len(ordered_fixations)) 

547 contiguous = np.concatenate(([0.0], np.cumsum(dur)[:-1])) if len(dur) else dur 

548 if "timestamp_ms" in ordered_fixations.columns: 

549 ts = pd.to_numeric(ordered_fixations["timestamp_ms"], errors="coerce").to_numpy( 

550 dtype=float 

551 ) 

552 total_dwell = float(dur.sum()) if len(dur) else 0.0 

553 if ( 

554 len(ts) 

555 and not np.isnan(ts).any() 

556 and (ts[-1] - ts[0]) >= REAL_TIMESTAMP_DWELL_FRAC * total_dwell 

557 ): 

558 return ts - ts[0] 

559 return contiguous

571def _in_any_box(fix_chunk: pd.DataFrame, word_chunk: pd.DataFrame) -> np.ndarray: 

572 """Boolean array (aligned to fix_chunk order): is each fixation inside any box?""" 

573 x0, y0, x1, y1 = word_box_bounds(word_chunk) 

574 fx = pd.to_numeric(fix_chunk["x"], errors="coerce").to_numpy(dtype=float) 

575 fy = pd.to_numeric(fix_chunk["y"], errors="coerce").to_numpy(dtype=float) 

576 inside = ( 

577 (fx[:, None] >= x0[None, :]) 

578 & (fx[:, None] <= x1[None, :]) 

579 & (fy[:, None] >= y0[None, :]) 

580 & (fy[:, None] <= y1[None, :]) 

581 ) 

582 return inside.any(axis=1)

585def cluster_word_lines(words: pd.DataFrame, tol_frac: float = 0.5) -> pd.Series: 

586 """Assign each word a 0-based line id by clustering on vertical position. 

587 

588 OneStop IA exports rarely carry a real per-word line number (the 

589 normalized ``line_idx`` is often a constant), so we infer visual lines from 

590 word-box geometry: words are sorted by vertical center and a new line 

591 starts whenever the center jumps by more than ``tol_frac`` of the median 

592 word height. Lines are numbered top-to-bottom starting at 0. 

593 

594 Returns an int Series aligned to ``words.index`` (empty when ``words`` is 

595 empty). Mirrors the line clustering used by ``plots.build_critical_span_overlay``. 

596 """ 

597 if words.empty: 

598 return pd.Series([], dtype="int64", index=words.index) 

599 heights = pd.to_numeric(words["height"], errors="coerce") 

600 typical_h = float(heights.median()) if heights.notna().any() else 1.0 

601 typical_h = typical_h if typical_h > 0 else 1.0 

602 y_center = pd.to_numeric(words["y"], errors="coerce") + heights.fillna(0) / 2.0 

603 order = y_center.sort_values(kind="stable") 

604 line_of_sorted = ( 

605 (order.diff().fillna(0) > typical_h * tol_frac).cumsum().astype(int) 

606 ) 

607 return line_of_sorted.reindex(words.index)

610def _groupwise(fixations: pd.DataFrame, words: pd.DataFrame, fn) -> pd.Series: 

611 """Apply ``fn(fix_chunk, word_chunk)`` per (participant, trial), aligning 

612 the per-chunk result back onto a Series indexed like ``fixations``. 

613 

614 Falls back to a single group when the id columns are absent (e.g. the 

615 figure builders pass already-sliced single-trial frames).""" 

616 keys = grouping_columns(fixations) 

617 # object dtype so a chunk fn may return bools (in-text) or floats (line ids) 

618 # without triggering a dtype-incompatibility cast; callers coerce the result. 

619 out = pd.Series(np.nan, index=fixations.index, dtype=object) 

620 has_groups = bool(keys) and keys == grouping_columns(words) 

621 if not has_groups: 

622 out.loc[:] = fn(fixations, words) 

623 return out 

624 word_groups = words.groupby(keys, sort=False) 

625 for key, fix_chunk in fixations.groupby(keys, sort=False): 

626 try: 

627 word_chunk = word_groups.get_group(key) 

628 except KeyError: 

629 continue 

630 if word_chunk.empty: 

631 continue 

632 out.loc[fix_chunk.index] = fn(fix_chunk, word_chunk) 

633 return out

636def fixation_in_text_mask(fixations: pd.DataFrame, words: pd.DataFrame) -> pd.Series: 

637 """Boolean Series: True where a fixation falls inside any word box. 

638 

639 "Out-of-text" fixations are simply ``~fixation_in_text_mask(...)``. Works 

640 on multi-trial frames (grouped by participant/trial) or on a single 

641 already-sliced trial. Fixations with non-finite coordinates count as 

642 out-of-text (mask = False).""" 

643 if fixations.empty: 

644 return pd.Series([], dtype=bool, index=fixations.index) 

645 if words.empty: 

646 return pd.Series(False, index=fixations.index) 

647 res = _groupwise(fixations, words, _in_any_box) 

648 return res.where(res.notna(), other=False).astype(bool)

651def assign_fixation_lines(fixations: pd.DataFrame, words: pd.DataFrame) -> pd.Series: 

652 """Assign each fixation the 0-based line id of the nearest text line. 

653 

654 Lines are derived from word geometry via :func:`cluster_word_lines`; each 

655 fixation is mapped to the line whose mean vertical center is closest to the 

656 fixation's y. Returns a float Series (NaN where unmappable) aligned to 

657 ``fixations.index`` so it can be used as a categorical color field.""" 

658 if fixations.empty: 

659 return pd.Series([], dtype="float64", index=fixations.index) 

660 if words.empty: 

661 return pd.Series(np.nan, index=fixations.index, dtype="float64") 

662 

663 def _nearest_line(fix_chunk: pd.DataFrame, word_chunk: pd.DataFrame) -> np.ndarray: 

664 lines = cluster_word_lines(word_chunk) 

665 y_center = ( 

666 pd.to_numeric(word_chunk["y"], errors="coerce") 

667 + pd.to_numeric(word_chunk["height"], errors="coerce").fillna(0) / 2.0 

668 ) 

669 centers = y_center.groupby(lines).mean() 

670 line_ids = centers.index.to_numpy(dtype=float) 

671 line_cy = centers.to_numpy(dtype=float) 

672 fy = pd.to_numeric(fix_chunk["y"], errors="coerce").to_numpy(dtype=float) 

673 dist = np.abs(fy[:, None] - line_cy[None, :]) 

674 nearest = np.where(np.isnan(fy), -1, dist.argmin(axis=1)) 

675 result = np.where(nearest >= 0, line_ids[np.clip(nearest, 0, None)], np.nan) 

676 return result 

677 

678 return pd.to_numeric(_groupwise(fixations, words, _nearest_line), errors="coerce")

681def compute_per_word_measures( 

682 fixations: pd.DataFrame, words: pd.DataFrame 

683) -> pd.DataFrame: 

684 """Compute canonical reading measures per word. 

685 

686 Returns a copy of `words` with computed columns added. Existing values on 

687 `words` (e.g. EyeLink IA metrics) take precedence over computed ones. 

688 """ 

689 if words.empty: 

690 return words.copy() 

691 

692 enriched = ( 

693 enrich_fixations(assign_fixations_to_words(fixations, words), words) 

694 if not fixations.empty 

695 else fixations 

696 ) 

697 

698 out = words.copy() 

699 key_cols = grouping_columns(words, include_word=True) 

700 # VAL-5: the letter scale for the landing measures, resolved once on the 

701 # whole frame (tiling is a property of the layout, so it must not be 

702 # re-detected per trial word, which is a subset with holes in it). 

703 char_advance = pd.Series(word_char_advance(words), index=words.index) 

704 

705 # Initialize defaults 

706 computed = ( 

707 words[key_cols] 

708 .drop_duplicates() 

709 .assign( 

710 _first_fixation_ms=np.nan, 

711 _first_pass_gaze_ms=np.nan, 

712 _regression_path_ms=np.nan, 

713 _total_fixation_ms=0.0, 

714 _n_fixations=0, 

715 _skip_flag=True, 

716 _regression_in_flag=False, 

717 _regression_out_flag=False, 

718 _first_fix_x=np.nan, 

719 _first_fix_y=np.nan, 

720 _initial_landing_position=np.nan, 

721 _initial_landing_distance=np.nan, 

722 _number_of_regressions_in=0, 

723 _second_pass_duration=0.0, 

724 _single_fixation_duration=np.nan, 

725 ) 

726 ) 

727 

728 analysis_fixations = enriched 

729 if "excluded" in enriched.columns: 

730 analysis_fixations = enriched[~enriched["excluded"].fillna(False).astype(bool)] 

731 if not analysis_fixations.empty and "word_id" in analysis_fixations.columns: 

732 per_word_rows = [] 

733 # Group fixations by trial to walk them in temporal order. 

734 trial_keys = grouping_columns(analysis_fixations) 

735 for group_key, fix_chunk in analysis_fixations.dropna( 

736 subset=["word_id"] 

737 ).groupby(trial_keys, sort=False): 

738 values = group_key if isinstance(group_key, tuple) else (group_key,) 

739 identity = dict(zip(trial_keys, values)) 

740 fix_chunk = fix_chunk.sort_values("timestamp_ms") 

741 # Total / n / first-fixation are per-word aggregations 

742 grp = fix_chunk.groupby("word_id") 

743 tot = grp["duration_ms"].sum() 

744 n = grp.size() 

745 ffd = grp["duration_ms"].first() 

746 ffx = grp["x"].first() 

747 ffy = grp["y"].first() 

748 second_pass = ( 

749 fix_chunk[ 

750 pd.to_numeric(fix_chunk.get("word_run"), errors="coerce").eq(2) 

751 ] 

752 .groupby("word_id")["duration_ms"] 

753 .sum() 

754 ) 

755 first_pass_counts = ( 

756 fix_chunk[ 

757 pd.to_numeric(fix_chunk.get("word_run"), errors="coerce").eq(1) 

758 ] 

759 .groupby("word_id") 

760 .size() 

761 ) 

762 

763 # First-pass gaze: walk the trial in order, accumulate runs. 

764 first_pass_gaze: dict[float, float] = {} 

765 regression_path: dict[float, float] = {} 

766 regression_in: set = set() 

767 regression_out: set = set() 

768 regression_in_count: dict[float, int] = {} 

769 

770 running_max = -np.inf 

771 current_run_word: float | None = None 

772 current_run_duration: float = 0.0 

773 # For regression-path: from first entry into a word until first 

774 # fixation past it, sum all durations. 

775 first_entry_seen: set = set() 

776 rp_open_for: dict[float, float] = {} 

777 

778 prev_word: float | None = None 

779 for row in fix_chunk.itertuples(): 

780 w = float(row.word_id) 

781 dur = float(row.duration_ms) 

782 

783 # First-pass gaze duration: continuous run on this word 

784 # starting from first entry, ending the first time we leave. 

785 if w not in first_pass_gaze: 

786 if current_run_word == w: 

787 current_run_duration += dur 

788 else: 

789 if current_run_word is not None: 

790 first_pass_gaze.setdefault( 

791 current_run_word, current_run_duration 

792 ) 

793 current_run_word = w 

794 current_run_duration = dur 

795 else: 

796 # Already past first pass; reset run tracker. 

797 if ( 

798 current_run_word is not None 

799 and current_run_word not in first_pass_gaze 

800 ): 

801 first_pass_gaze.setdefault( 

802 current_run_word, current_run_duration 

803 ) 

804 current_run_word = None 

805 current_run_duration = 0.0 

806 

807 # Regression-path: from the first entry into a word, sum 

808 # durations until the next fixation lands on a strictly later 

809 # word. 

810 if w not in first_entry_seen: 

811 first_entry_seen.add(w) 

812 rp_open_for[w] = dur 

813 else: 

814 for k in list(rp_open_for.keys()): 

815 if w <= k: 

816 # Still within or back-tracking; keep accumulating. 

817 rp_open_for[k] += dur 

818 # Close any open RP windows for words we've now moved past. 

819 for k in list(rp_open_for.keys()): 

820 if w > k and k != w: 

821 regression_path.setdefault(k, rp_open_for.pop(k)) 

822 

823 # Regression-in: stepping back to an earlier word counts the 

824 # destination as receiving an in-regression. 

825 if prev_word is not None and w < prev_word: 

826 regression_in.add(w) 

827 regression_out.add(prev_word) 

828 regression_in_count[w] = regression_in_count.get(w, 0) + 1 

829 

830 running_max = max(running_max, w) 

831 prev_word = w 

832 

833 # Flush remaining first-pass run 

834 if current_run_word is not None and current_run_word not in first_pass_gaze: 

835 first_pass_gaze[current_run_word] = current_run_duration 

836 # Flush remaining regression-path windows (reader never moved past) 

837 for k, v in rp_open_for.items(): 

838 regression_path.setdefault(k, v) 

839 

840 for w in tot.index: 

841 word_mask = pd.to_numeric(words["word_id"], errors="coerce") == w 

842 for column, value in identity.items(): 

843 word_mask &= words[column] == value 

844 trial_word = words[word_mask] 

845 landing_position = landing_distance = np.nan 

846 if not trial_word.empty: 

847 target = trial_word.iloc[0] 

848 text_len = max(len(str(target.get("text", ""))), 1) 

849 width = float(pd.to_numeric(target.get("width"), errors="coerce")) 

850 char_width = float(char_advance.loc[trial_word.index[0]]) 

851 if np.isfinite(char_width) and char_width > 0 and width > 0: 

852 rtl = bool(target.get("right_to_left", False)) 

853 # BUG-27: RTL counts from where the glyphs *end* 

854 # (`x + n × advance`), not from the padded box edge 

855 # `x + width` — on a tiling layout those are one whole 

856 # advance apart, so an RTL landing read a letter late 

857 # and disagreed with `aggregation.landing_positions`, 

858 # which measures across the glyph run on both sides. 

859 offset = ( 

860 float(target.get("x")) 

861 + text_len * char_width 

862 - float(ffx.loc[w]) 

863 if rtl 

864 else float(ffx.loc[w]) - float(target.get("x")) 

865 ) 

866 landing_position = offset / char_width + 1.0 

867 landing_distance = landing_position - (text_len + 1) / 2.0 

868 per_word_rows.append( 

869 dict( 

870 **identity, 

871 word_id=w, 

872 _first_fixation_ms=float(ffd.loc[w]), 

873 _first_pass_gaze_ms=float(first_pass_gaze.get(w, np.nan)), 

874 _regression_path_ms=float( 

875 regression_path.get(w, np.nan) 

876 if w in first_entry_seen 

877 else np.nan 

878 ), 

879 _total_fixation_ms=float(tot.loc[w]), 

880 _n_fixations=int(n.loc[w]), 

881 _skip_flag=bool(np.isnan(first_pass_gaze.get(w, np.nan))), 

882 _regression_in_flag=w in regression_in, 

883 _regression_out_flag=w in regression_out, 

884 _first_fix_x=float(ffx.loc[w]), 

885 _first_fix_y=float(ffy.loc[w]), 

886 _initial_landing_position=landing_position, 

887 _initial_landing_distance=landing_distance, 

888 _number_of_regressions_in=int(regression_in_count.get(w, 0)), 

889 _second_pass_duration=float(second_pass.get(w, 0.0)), 

890 _single_fixation_duration=( 

891 float(ffd.loc[w]) 

892 if int(first_pass_counts.get(w, 0)) == 1 

893 else np.nan 

894 ), 

895 ) 

896 ) 

897 

898 if per_word_rows: 

899 new_df = pd.DataFrame(per_word_rows) 

900 updated = computed.merge( 

901 new_df, on=key_cols, how="left", suffixes=("_default", "") 

902 ) 

903 for col in new_df.columns: 

904 if col in key_cols: 

905 continue 

906 default_col = f"{col}_default" 

907 if default_col in updated.columns: 

908 updated[col] = updated[col].where( 

909 updated[col].notna(), updated[default_col] 

910 ) 

911 updated = updated.drop(columns=default_col) 

912 computed = updated 

913 

914 out = out.merge(computed, on=key_cols, how="left") 

915 

916 # Map computed -> canonical name, keeping any existing value. 

917 rename_map = { 

918 "_first_fixation_ms": "first_fixation_ms", 

919 "_first_pass_gaze_ms": "first_pass_gaze_duration_ms", 

920 "_regression_path_ms": "regression_path_duration_ms", 

921 "_total_fixation_ms": "total_fixation_duration_ms", 

922 "_n_fixations": "n_fixations", 

923 "_skip_flag": "skip_flag", 

924 "_regression_in_flag": "regression_in_flag", 

925 "_regression_out_flag": "regression_out_flag", 

926 "_first_fix_x": "first_fix_x", 

927 "_first_fix_y": "first_fix_y", 

928 "_initial_landing_position": "initial_landing_position", 

929 "_initial_landing_distance": "initial_landing_distance", 

930 "_number_of_regressions_in": "number_of_regressions_in", 

931 "_second_pass_duration": "second_pass_duration_ms", 

932 "_single_fixation_duration": "single_fixation_duration_ms", 

933 } 

934 for src, dst in rename_map.items(): 

935 if src not in out.columns: 

936 continue 

937 if dst in out.columns: 

938 out[dst] = out[dst].where(out[dst].notna(), out[src]) 

939 else: 

940 out[dst] = out[src] 

941 out = out.drop(columns=src) 

942 

943 # Canonical aliases used elsewhere in the app 

944 if ( 

945 "first_pass_gaze_duration_ms" in out.columns 

946 and "gaze_duration_ms" not in out.columns 

947 ): 

948 out["gaze_duration_ms"] = out["first_pass_gaze_duration_ms"] 

949 

950 # Ensure dtypes 

951 for col in ["n_fixations"]: 

952 if col in out.columns: 

953 out[col] = ( 

954 pd.to_numeric(out[col], errors="coerce").fillna(0).astype("Int64") 

955 ) 

956 for col in ["skip_flag", "regression_in_flag", "regression_out_flag"]: 

957 if col in out.columns: 

958 out[col] = ( 

959 out[col].astype(object).where(out[col].notna(), False).astype(bool) 

960 ) 

961 

962 return out
