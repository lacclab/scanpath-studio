1"""Loaders for public eye-tracking-while-reading corpora. 

2 

3Currently: PoTeC (Potsdam Textbook Corpus, Jakobi et al. 2024, 

4https://github.com/DiLi-Lab/PoTeC) — a German corpus of 75 readers × 12 

5textbook texts — and MultiplEYE (https://multipleye.eu), a large multilingual 

6corpus whose per-session, filename-keyed shape :func:`load_multipleye` adapts 

7to the generic pipeline. Between them they exercise the two dataset shapes the 

8generic pipeline supports: 

9 

10* **multi-file** — fixations ship as one TSV per reader × text 

11 (``reader0_b0_scanpath.tsv`` … 900 files), concatenated on load; 

12* **stimulus-level AoIs** — word bounding boxes ship once per *text* with no 

13 participant column, and are broadcast across readers 

14 (``data.broadcast_stimulus_words``). 

15 

16PoTeC fixations carry no pixel coordinates — only the fixated character's 

17index. The loader reconstructs (x, y) as the center of that character's 

18bounding box from the per-text ``.ias`` AOI files, giving within-word landing 

19positions. (For AOI-sequence datasets *without* character AOIs, the generic 

20fallback places fixations at word-box centers instead.) 

21 

22Typical use:: 

23 

24 from scanpath_studio.datasets import load_potec 

25 

26 words, fixations = load_potec("data/PoTeC", download=True) 

27 fig = scanpath_studio.plot_scanpath(words, fixations, participant="0", trial="b0") 

28"""

87def download_potec(root, *, fixation_source: str = "scanpaths") -> Path: 

88 """Download the PoTeC files :func:`load_potec` needs into ``root``. 

89 

90 Fetches the per-trial eye-tracking archive (~45 MB zip) from PoTeC's OSF 

91 repository and the 24 per-text AOI files (word boxes + character boxes) 

92 from the PoTeC GitHub repo. Skips anything already present, so it's safe 

93 to call repeatedly (and it's a no-op on a full clone of the PoTeC repo 

94 where ``download_data_files.py`` has been run). 

95 

96 ``fixation_source`` is ``"scanpaths"`` (default; temporally ordered 

97 fixations with word indices) or ``"fixations"``. 

98 """ 

99 if fixation_source not in _POTEC_OSF_RESOURCES: 

100 raise ValueError( 

101 f"fixation_source must be one of {sorted(_POTEC_OSF_RESOURCES)}, " 

102 f"got {fixation_source!r}" 

103 ) 

104 root = Path(root) 

105 

106 eyetracking_dir = root / "eyetracking_data" / fixation_source 

107 if not eyetracking_dir.is_dir(): 

108 url = _POTEC_OSF_URL.format(resource=_POTEC_OSF_RESOURCES[fixation_source]) 

109 print(f"Downloading PoTeC {fixation_source} from {url} …") 

110 with urllib.request.urlopen(url) as response: 

111 payload = response.read() 

112 (root / "eyetracking_data").mkdir(parents=True, exist_ok=True) 

113 with zipfile.ZipFile(io.BytesIO(payload)) as archive: 

114 members = [ 

115 m 

116 for m in archive.namelist() 

117 # The OSF zips carry macOS resource-fork cruft; keep only the 

118 # real per-trial TSVs. 

119 if m.startswith(f"{fixation_source}/") and m.endswith(".tsv") 

120 ] 

121 archive.extractall(root / "eyetracking_data", members=members) 

122 

123 for text_id in _POTEC_TEXTS: 

124 for rel in ( 

125 f"stimuli/word_aoi_texts/word_aoi_{text_id}.tsv", 

126 f"stimuli/aoi_texts/{text_id}.ias", 

127 ): 

128 dest = root / rel 

129 if dest.is_file(): 

130 continue 

131 dest.parent.mkdir(parents=True, exist_ok=True) 

132 url = _POTEC_RAW_URL.format(path=rel) 

133 print(f"Downloading {url} …") 

134 # Write to a temp file and atomically rename into place, so an 

135 # interrupted fetch never leaves a truncated AOI file that 

136 # `dest.is_file()` / `potec_present` would then treat as complete — 

137 # mirroring download_onestop. 

138 tmp = dest.with_name(dest.name + ".part") 

139 with urllib.request.urlopen(url) as response: 

140 tmp.write_bytes(response.read()) 

141 tmp.replace(dest) 

142 return root

145def potec_present(root) -> bool: 

146 """True when ``root`` already holds the full PoTeC corpus a load needs. 

147 

148 The app loads *every* text, so this requires the per-text word + character 

149 AOI files for **all** texts (exactly what :func:`download_potec` fetches), 

150 plus a populated eye-tracking folder. A lenient "any AOI file" check would 

151 pass a partial tree and then crash mid-load (`_potec_words` raises on the 

152 first missing text) with no way to recover; requiring all of them instead 

153 lets the app offer the Download button, which self-heals the gap. Cheap 

154 (path stats only), so the status shows without reading any CSV.""" 

155 root = Path(root) 

156 base = root / "eyetracking_data" 

157 has_fixations = any( 

158 (base / s).is_dir() and any((base / s).glob("*.tsv")) 

159 for s in ("scanpaths", "fixations") 

160 ) 

161 word_dir = root / "stimuli" / "word_aoi_texts" 

162 char_dir = root / "stimuli" / "aoi_texts" 

163 has_all_aoi = all( 

164 (word_dir / f"word_aoi_{text_id}.tsv").is_file() 

165 and (char_dir / f"{text_id}.ias").is_file() 

166 for text_id in _POTEC_TEXTS 

167 ) 

168 return has_fixations and has_all_aoi

171def _potec_words(root: Path, texts: Iterable[str]) -> pd.DataFrame: 

172 """Stimulus-level word table: one row per word per text, with boxes. 

173 

174 PoTeC keys the text id in the AOI *filename* only; it becomes a regular 

175 ``text_id`` column here. Line indices come from the character-level 

176 ``.ias`` files (the word AOI files don't carry them) via the lines' y 

177 positions.""" 

178 frames = [] 

179 for text_id in texts: 

180 path = root / "stimuli" / "word_aoi_texts" / f"word_aoi_{text_id}.tsv" 

181 if not path.is_file(): 

182 raise FileNotFoundError( 

183 f"PoTeC word AOI file not found: {path} — pass download=True " 

184 "or run PoTeC's download_data_files.py in a repo clone." 

185 ) 

186 words = _read_potec_tsv(path) 

187 words["text_id"] = text_id 

188 

189 ias = _read_potec_ias(root, text_id) 

190 # Character boxes on the same text line share start_y, so the char 

191 # AOIs give an exact y → line lookup for the word boxes. 

192 y_to_line = ias.drop_duplicates("start_y").set_index("start_y")["line"] 

193 words["line"] = words["start_y"].map(y_to_line) 

194 frames.append(words) 

195 return pd.concat(frames, ignore_index=True)

208def _potec_fixations( 

209 root: Path, 

210 texts: Iterable[str], 

211 readers: Iterable | None = None, 

212) -> pd.DataFrame: 

213 """Concatenated per-trial fixation files with reconstructed coordinates. 

214 

215 Prefers ``eyetracking_data/scanpaths/`` (fixations in temporal order, with 

216 word indices) and falls back to ``eyetracking_data/fixations/``. Each 

217 fixation's (x, y) is the center of the fixated character's box from the 

218 per-text ``.ias`` file — PoTeC discards the original screen coordinates.""" 

219 base = root / "eyetracking_data" 

220 source = next((s for s in ("scanpaths", "fixations") if (base / s).is_dir()), None) 

221 if source is None: 

222 raise FileNotFoundError( 

223 f"No PoTeC fixation data under {base} — expected a 'scanpaths' or " 

224 "'fixations' folder. Pass download=True, or run PoTeC's " 

225 "download_data_files.py in a repo clone." 

226 ) 

227 suffix = "scanpath" if source == "scanpaths" else "fixations" 

228 

229 reader_set = None if readers is None else {str(r) for r in readers} 

230 frames = [] 

231 for text_id in texts: 

232 char_boxes = _read_potec_ias(root, text_id) 

233 char_x = (char_boxes["start_x"] + char_boxes["end_x"]) / 2.0 

234 char_y = (char_boxes["start_y"] + char_boxes["end_y"]) / 2.0 

235 centers = pd.DataFrame( 

236 {"aoi": char_boxes["aoi"], "x": char_x, "y": char_y} 

237 ).drop_duplicates("aoi") 

238 

239 for path in sorted((base / source).glob(f"reader*_{text_id}_{suffix}.tsv")): 

240 reader_id = path.stem.removeprefix("reader").split("_")[0] 

241 if reader_set is not None and reader_id not in reader_set: 

242 continue 

243 fixations = _read_potec_tsv(path) 

244 fixations = fixations.merge(centers, on="aoi", how="left") 

245 frames.append(fixations) 

246 if not frames: 

247 raise FileNotFoundError( 

248 f"No PoTeC fixation files matched the requested readers/texts " 

249 f"under {base / source}." 

250 ) 

251 return pd.concat(frames, ignore_index=True, sort=False)

359# OneStop Eye Movements — 360-participant English corpus (Berzak et al. 2025, 

360# https://github.com/lacclab/OneStop-Eye-Movements). Distributed on OSF as 

361# interest-area (word) + fixation reports, split by reading **regime** and by 

362# trial **part** (which screen of a trial — title / question preview / 

363# paragraph / questions / answers / QA / feedback). The reports share the 

364# bundled demo's schema (the demo is a 3-pid subset of the Paragraph part), so 

365# the generic auto-detect → normalize pipeline handles every part with no 

366# part-specific column mapping — this loader only fetches, reads, and (when more 

367# than one part is loaded) folds the part into the trial identity so the parts 

368# don't collide. 

369# 

370# Two **variants**: 

371# * ``public`` — the OSF download-on-demand release (this module fetches it). 

372# * ``lacclab`` — a lab-processed local export with ~40 extra derived columns 

373# (``unique_paragraph_id``, span indices, normalized dwell, …); a superset 

374# of the public schema, so it flows through the same pipeline (and its 

375# ``unique_paragraph_id`` wins in normalization). No download — a local path. 

376# 

377# Distinct from the env-var "OneStop server bundle" source 

378# (``data.load_onestop_server_bundle``), which serves a lacclab export via 

379# ``$ONESTOP_DATA_DIR`` and its per-pid shards for review-app deep links. 

380# ---------------------------------------------------------------------------

515def download_onestop( 

516 root, 

517 *, 

518 regime: str = "ordinary", 

519 parts: Iterable[str] | None = None, 

520) -> Path: 

521 """Download a OneStop regime + parts' IA + fixation reports into ``root``. 

522 

523 Fetches the two CSV.zip reports for each chosen ``part`` from OneStop's OSF 

524 release into ``root``, skipping any already present, so it's safe to call 

525 repeatedly. The reports are large (tens to hundreds of MB each); caching them 

526 on disk means only the first load pays the download. 

527 

528 ``regime`` is one of ``ordinary``, ``information_seeking``, ``repeated``, 

529 ``information_seeking_repeated``. ``parts`` is any subset of 

530 ``Title / Question_Preview / Paragraph / Questions / Answers / QA / 

531 Feedback`` (default: just Paragraph). Only Paragraph is regime-split on OSF; 

532 the other parts come from the all-regimes full release. 

533 

534 (Download is a *public*-variant operation — the lacclab variant is a local 

535 export with no download URL.) 

536 """ 

537 if regime not in _ONESTOP_REGIMES: 

538 raise ValueError( 

539 f"regime must be one of {sorted(_ONESTOP_REGIMES)}, got {regime!r}" 

540 ) 

541 root = Path(root) 

542 root.mkdir(parents=True, exist_ok=True) 

543 for part in _normalize_onestop_parts(parts): 

544 for kind in ("ia", "fixations"): 

545 dest = _onestop_report_path(root, kind, regime, part) 

546 if dest.is_file(): 

547 continue

554 # interrupted write (killed process / full disk) never leaves a 

555 # truncated .csv.zip that `dest.is_file()` would then skip forever — 

556 # forcing a manual delete. The reports are large, so the window is real.

564# BUG-43: the public OSF release ships **no** ``unique_paragraph_id``, so the 

565# only text identity on its reports is ``paragraph_id`` — the paragraph's index 

566# *within its article*, 1..7. On its own that makes paragraph 3 of article 1 and 

567# paragraph 3 of article 27 the same "text" (162 paragraphs read as 7), and 

568# because the same id then repeats dozens of times per reader, 

569# ``data._disambiguate_repeated_readings`` ranks the collisions apart into trial 

570# ids like ``3_r17`` — an order, not an identity.

572# So compose the ids the corpus's own tooling writes, in the same field order 

573# ``update_sample_data.add_unique_ids`` uses for the bundled demo. The demo is a 

574# subset of this corpus, so a trial must not change identity depending on which 

575# of the two was opened. The level is part of the *text* id there, and stays 

576# one here: OneStop's Advanced and Elementary renderings of a paragraph are

592def _compose_onestop_ids(frame: pd.DataFrame) -> pd.DataFrame: 

593 """Give an OneStop report the ``unique_paragraph_id`` / ``unique_trial_id`` 

594 the public release omits (BUG-43). 

595 

596 Each id is composed only when the frame does not already carry it: the 

597 lacclab export ships its own ``unique_paragraph_id`` (in its own field 

598 order), and that is the publisher's identity — the trial id is composed 

599 *under* it rather than over it. A frame missing any of the composing 

600 columns is returned untouched: there is nothing to invent an id from. 

601 """ 

602 from . import data 

603 

604 columns = set(frame.columns) 

605 if "unique_paragraph_id" not in columns: 

606 if not set(_ONESTOP_PARAGRAPH_ID_PARTS) <= columns: 

607 return frame 

608 frame = frame.copy() 

609 frame["unique_paragraph_id"] = _join_ids(frame, _ONESTOP_PARAGRAPH_ID_PARTS) 

610 columns = set(frame.columns) 

611 if "unique_trial_id" not in columns: 

612 if not set(_ONESTOP_TRIAL_ID_PARTS) <= columns: 

613 return frame 

614 frame = frame.copy() 

615 reading = data.coerce_flag(frame["repeated_reading_trial"]).astype(int) 

616 frame["unique_trial_id"] = ( 

617 frame["participant_id"].astype(str) 

618 + "_" 

619 + frame["unique_paragraph_id"].astype(str) 

620 + "_r" 

621 + reading.astype(str) 

622 ) 

623 return frame

626def _join_ids(frame: pd.DataFrame, columns: Iterable[str]) -> pd.Series: 

627 """``"_"``-joined string form of ``columns`` — one composed id per row.""" 

628 parts = [frame[column].astype(str) for column in columns] 

629 joined = parts[0] 

630 for part in parts[1:]: 

631 joined = joined + "_" + part

638 """Read one part's report and stamp a ``part`` column onto it. 

639 

640 QA repeats the question words in the answer region with the *same* IA_ID, so 

641 its per-trial word ids aren't unique — drop the exact-duplicate rows so the 

642 fixation→word assignment (which keys on word id) keeps one box per word."""

644

787# MultiplEYE — multilingual eye-tracking-while-reading corpus 

788# (https://multipleye.eu). Tested against the read-only ZH/Chinese Zurich 

789# sample under ``data/MultiplEYE_ZH_CH_Zurich_1_2025``. 

790# --------------------------------------------------------------------------- 

791# 

792# Why a dedicated loader (the generic Upload flow can't do this on its own): 

793# 

794# * **Identity is in the path, not the columns.** Per-session folders 

795# ``{pid}_ZH_CH_1_ET{1|2}/`` (under ``fixations/`` and ``scanpaths/``) hold 

796# one comma-CSV per trial; none of the files carry a participant / trial / 

797# stimulus column. We parse all four from the folder + file name. 

798# * **ET1 and ET2 read disjoint stimuli**, so a *reader* is the whole session 

799# string — ``participant_id = "001_ZH_CH_1_ET1"`` — not the bare pid. 

800# * **One stimulus spans several screens** ``page_1..page_N`` (plus the 

801# comprehension-question screens that follow them) which all reuse the *same* 

802# on-screen coordinates. DATA-24 models that directly: one reading of one 

803# stimulus is **one trial** (``trial_id == text_id == "<stimulus>"``) whose 

804# screens are the corpus's own ``page`` values — ``screen_id = "page_1"`` / 

805# ``"question_4111"`` — each its own coordinate space (``multipart.py``). 

806# ``screen_index`` is ranked from **that reader's own fixation onsets**, never 

807# from the screen name: reading pages are shown in page order but the question 

808# order is *shuffled per reader*. ``screen_kind`` (``reading`` | ``question``) 

809# labels the two. ``familiarity_rating_screen_*`` / ``subject_difficulty_screen`` 

810# stay out: the corpus ships no AOI file for them, and 

811# ``multipart.validate_matching_parts`` rejects word-box-less screens by design. 

812# * **Word boxes ship once per stimulus** (no participant) as *character*-level 

813# AOI files ``stimuli_*/aoi_stimuli_*/<stimulus>_aoi.csv``. We aggregate chars 

814# to one bounding box per (page, word_idx). ``word_idx`` is unique within a 

815# page — hence within a screen — so it is the word id directly. The 

816# **question** screens' boxes come from ``<stimulus>_aoi_questions.csv``, whose 

817# rows are per *answer-layout version*: which version a reader saw is looked up 

818# in ``stimuli_*/config/stimulus_order_versions_*.csv`` by their bare pid, so 

819# question boxes are always per reader. 

820# 

821# Fixation source: ``scanpaths/`` (preferred — already page/word-tagged) or 

822# ``fixations/`` (raw onset/duration/x/y/page, no word linkage). The 

823# ``scanpaths/`` export is pre-filtered to reading pages, so **question screens 

824# are always read from ``fixations/``** whatever ``fixation_source`` says; the 

825# resulting mixed provenance inside one trial is visible as ``screen_kind``. 

826 

827MULTIPLEYE_FIXATION_SOURCES = ("scanpaths", "fixations") 

828 

829# The screen kinds a MultiplEYE trial is modelled from, in presentation order. 

830MULTIPLEYE_SCREEN_KINDS = ("reading", "question")

833# (``Monitor_resolution_in_px`` / ``RESOLUTION``) — the physical screen the data 

834# was recorded on; pass as ``canvas_size`` to :func:`scanpath_studio.plot_scanpath` 

835# for true-to-scale rendering.

839# and was shown **centered**. The raw AOI/fixation coords are image-relative 

840# (text starts at the image's 81/88 px margins), so the loader shifts them by the 

841# centering offset below → they land where the participant actually saw them on

951def _multipleye_font_css(font_file: str) -> str: 

952 """CSS font-family stack for a config ``FONT`` path (e.g. a ``.ttf`` filename).""" 

953 stem = re.sub(r"[^a-z0-9]", "", Path(font_file).stem.lower()) 

954 for drop in _FONT_NAME_DROP: 

955 stem = stem.removesuffix(drop) 

956 if stem in _MULTIPLEYE_FONT_CSS: 

957 return _MULTIPLEYE_FONT_CSS[stem] 

958 # Unknown font: humanise the file stem (spaces at case/digit boundaries) and 

959 # append a monospace fallback, plus a CJK fallback when the name implies CJK. 

960 raw = Path(font_file).stem 

961 human = re.sub(r"(?<=[a-z])(?=[A-Z])|(?<=[A-Za-z])(?=[0-9])", " ", raw).strip() 

962 tail = "'Noto Sans CJK SC', monospace" if "cjk" in stem else "monospace" 

963 return f"'{human}', {tail}" if human else tail

968 

969 Reads ``FONT_SIZE`` (monitor px the images were rendered at) and ``FONT`` (the 

970 typeface) from ``config_*.py`` by regex — the file imports lab-specific paths, 

971 so we never exec it. Returns ``(None, None)`` when no config / no FONT_SIZE. 

972 """ 

973 path = _multipleye_config_path(root) 

974 if path is None: 

975 return None, None

989def _multipleye_char_boxes(chars: pd.DataFrame, group: list) -> pd.DataFrame: 

990 """Aggregate character AOI rows to one bounding box per ``group``. 

991 

992 Emits *edge* columns (``left/right/top/bottom``) already shifted onto the 

993 centered stimulus' on-screen position, so they are true-to-scale on 

994 ``MULTIPLEYE_MONITOR``. Groups keep first-appearance order (the caller sorts 

995 the characters into reading order first).""" 

996 chars = chars.copy() 

997 chars["_right"] = chars["top_left_x"] + chars["width"] 

998 chars["_bottom"] = chars["top_left_y"] + chars["height"] 

999 boxes = ( 

1000 chars.groupby(group, sort=False) 

1001 .agg( 

1002 left=("top_left_x", "min"), 

1003 top=("top_left_y", "min"), 

1004 right=("_right", "max"), 

1005 bottom=("_bottom", "max"), 

1006 line_idx=("line_idx", "min"), 

1007 word=("word", "first"), 

1008 ) 

1009 .reset_index() 

1010 ) 

1011 off_x, off_y = _MULTIPLEYE_IMAGE_ORIGIN 

1012 boxes["left"] += off_x 

1013 boxes["right"] += off_x 

1014 boxes["top"] += off_y 

1015 boxes["bottom"] += off_y 

1016 return boxes

1131def _multipleye_question_boxes_from_frame( 

1132 chars: pd.DataFrame, stimulus: str, version: int | None 

1133) -> pd.DataFrame: 

1134 """Question-screen word boxes for one stimulus at one answer-layout version. 

1135 

1136 ``chars`` is a ``<stimulus>_aoi_questions.csv`` frame, whose ``page`` values 

1137 name the AOI block (``Lit_Alchemist_4_question_04111_target``) or the question 

1138 stem (``question_04111``) with the stimulus id **zero-padded** — the join back 

1139 to the fixations' ``question_4111`` is therefore on ``int(question_id)``. 

1140 

1141 Every block restarts ``word_idx`` at 0, so ``word_idx`` alone cannot be the 

1142 word id: the blocks are put in reading order and one counter runs across 

1143 them, with ``line_idx`` densely re-ranked the same way, so both are unique 

1144 within the screen. The block name survives as ``aoi_block``, a genuinely 

1145 useful per-word facet. Returns an empty frame when the file / version yields 

1146 nothing.""" 

1147 if chars is None or getattr(chars, "empty", True) or "page" not in chars.columns: 

1148 return pd.DataFrame() 

1149 if "question_image_version" in chars.columns: 

1150 if version is None: 

1151 return pd.DataFrame() 

1152 wanted = f"question_images_version_{int(version)}" 

1153 # Filter BEFORE copying: the file holds every layout version (250 in the 

1154 # ZH-CH sample), so this keeps well under 1% of the rows. 

1155 chars = chars[chars["question_image_version"].astype(str) == wanted] 

1156 parsed = chars["page"].map(_multipleye_question_parts) 

1157 chars = chars[parsed.notna()].copy() 

1158 if chars.empty: 

1159 return pd.DataFrame() 

1160 parsed = parsed[parsed.notna()] 

1161 chars["_question_id"] = [int(value[0]) for value in parsed] 

1162 chars["_aoi_block"] = [str(value[1]) for value in parsed] 

1163 

1164 order = _multipleye_question_block_order(chars) 

1165 sort_by = [ 

1166 column 

1167 for column in ("line_idx", "char_idx_in_line", "char_idx") 

1168 if column in chars.columns 

1169 ] 

1170 if sort_by: 

1171 chars = chars.sort_values(_QUESTION_KEY + sort_by, kind="stable") 

1172 # One aggregation for the whole stimulus/version rather than one per block: 

1173 # the groups are ~10 rows each, so per-call pandas overhead dominated. 

1174 out = _multipleye_char_boxes(chars, _QUESTION_KEY + ["word_idx"]) 

1175 out = out.merge(order, on=_QUESTION_KEY, how="left").sort_values( 

1176 ["_question_id", "_block_order", "line_idx", "word_idx"], kind="stable" 

1177 ) 

1178 per_question = out.groupby("_question_id", sort=False) 

1179 out["word_idx"] = per_question.cumcount() 

1180 # Dense per-screen line ids: the group codes run in the sorted order above, so 

1181 # subtracting each question's first code rebases them to 0. 

1182 codes = out.groupby(_QUESTION_KEY[:1] + ["_block_order", "line_idx"], sort=False) 

1183 line_key = codes.ngroup() 

1184 out["line_idx"] = line_key - line_key.groupby(out["_question_id"]).transform("min") 

1185 out["page"] = [_multipleye_question_screen_id(q) for q in out["_question_id"]] 

1186 out = out.rename(columns={"_question_id": "question_id", "_aoi_block": "aoi_block"}) 

1187 out["screen_kind"] = "question" 

1188 out = out.drop(columns=["_block_order"]).reset_index(drop=True) 

1189 return _multipleye_stamp_box_identity(out, stimulus)

1614def _multipleye_words_per_reader( 

1615 stim_boxes: pd.DataFrame, 

1616 rm: pd.DataFrame, 

1617 fixations: pd.DataFrame, 

1618 question_boxes: pd.DataFrame | None = None, 

1619) -> pd.DataFrame: 

1620 """Reading-page boxes per reader, scoped to the screens they actually fixated. 

1621 

1622 The stimulus-level boxes are replicated onto the ``(reader, stimulus, page)`` 

1623 triples present in that reader's fixations — an **inner** join, so a page the 

1624 reader skipped never becomes an orphan screen for 

1625 ``multipart.validate_matching_parts`` to reject. That reader's pre-aggregated 

1626 reading measures merge on ``(participant, stimulus, page, word_idx)`` 

1627 (``word_idx`` restarts per page, so page MUST be in the key). Question-screen 

1628 boxes are already per reader and are appended as they are.""" 

1629 if stim_boxes.empty: 

1630 words = stim_boxes 

1631 else: 

1632 pairs = fixations[["participant_id", "stimulus", "page"]].drop_duplicates() 

1633 words = pairs.merge(stim_boxes, on=["stimulus", "page"], how="inner") 

1634 if not rm.empty: 

1635 words = words.merge( 

1636 rm, on=["participant_id", "stimulus", "page", "word_idx"], how="left" 

1637 ) 

1638 if question_boxes is not None and not question_boxes.empty: 

1639 words = pd.concat([words, question_boxes], ignore_index=True, sort=False) 

1640 return words

1643def _multipleye_question_word_boxes( 

1644 aoi_source, fixations: pd.DataFrame, versions: dict 

1645) -> pd.DataFrame: 

1646 """Per-(reader, stimulus) question-screen word boxes. 

1647 

1648 ``aoi_source`` is either the corpus AOI directory (files are read on demand) 

1649 or a ``{stimulus -> question-AOI frame}`` map (the upload path). Which answer 

1650 layout a reader saw is looked up in ``versions`` by their **bare pid**, so a 

1651 reader with no assigned version — or a stimulus with no question-AOI rows — 

1652 contributes no boxes, and the caller drops those question screens rather than 

1653 guessing a layout that would draw plausible boxes in the wrong places.""" 

1654 if fixations.empty or "screen_kind" not in fixations.columns: 

1655 return pd.DataFrame() 

1656 wanted = fixations[fixations["screen_kind"] == "question"] 

1657 if wanted.empty: 

1658 return pd.DataFrame() 

1659 if not versions: 

1660 _LOGGER.warning( 

1661 "MultiplEYE: no answer-layout version table, so the %d question " 

1662 "screen(s) are skipped (their AOI layout is reader-specific).", 

1663 wanted[["participant_id", "trial_id", "page"]].drop_duplicates().shape[0], 

1664 ) 

1665 return pd.DataFrame() 

1666 

1667 # Two memos, because they have different granularities: the corpus assigns a 

1668 # DISTINCT layout version to every participant, so a per-(stimulus, version) 

1669 # box cache almost never hits — but the file it parses is per stimulus and is 

1670 # the expensive part (~15 MB each), so that read is memoized separately. 

1671 files: dict = {} 

1672 cache: dict = {} 

1673 frames = [] 

1674 for (reader, stimulus), group in wanted.groupby( 

1675 ["participant_id", "stimulus"], sort=False 

1676 ): 

1677 stimulus = str(stimulus) 

1678 version = versions.get(_multipleye_bare_pid(reader)) 

1679 key = (stimulus, version) 

1680 if key not in cache: 

1681 if version is None: 

1682 # No assigned layout: don't even read the file for it. 

1683 cache[key] = pd.DataFrame() 

1684 else: 

1685 if stimulus not in files: 

1686 files[stimulus] = _multipleye_question_aoi_frame( 

1687 aoi_source, stimulus 

1688 ) 

1689 cache[key] = _multipleye_question_boxes_from_frame( 

1690 files[stimulus], stimulus, version 

1691 ) 

1692 boxes = cache[key] 

1693 if boxes.empty: 

1694 _LOGGER.warning( 

1695 "MultiplEYE: no question AOI rows for reader %s on %s " 

1696 "(layout version %s) — its question screens are skipped.", 

1697 reader, 

1698 stimulus, 

1699 version, 

1700 ) 

1701 continue 

1702 seen = set(group["page"].astype(str)) 

1703 boxes = boxes[boxes["page"].isin(seen)].copy() 

1704 boxes["participant_id"] = reader 

1705 frames.append(boxes) 

1706 return ( 

1707 pd.concat(frames, ignore_index=True, sort=False) if frames else pd.DataFrame() 

1708 )

1719def _multipleye_drop_screens_without_boxes( 

1720 words: pd.DataFrame, fixations: pd.DataFrame 

1721) -> pd.DataFrame: 

1722 """Drop fixations on screens that have no word boxes, loudly. 

1723 

1724 ``multipart.validate_matching_parts`` rejects a screen present in one report 

1725 and absent from the other, so a screen we could not build boxes for (a 

1726 question screen whose layout version is unknown, a stimulus whose AOI file 

1727 was not uploaded) must be dropped here rather than crashing the whole load 

1728 inside ``data.harmonize_frames``.""" 

1729 if words.empty or fixations.empty or "page" not in words.columns: 

1730 return fixations 

1731 keys = ["trial_id", "page"] 

1732 if "participant_id" in words.columns: 

1733 keys.insert(0, "participant_id") 

1734 present = words[keys].drop_duplicates().astype(str) 

1735 present["_has_boxes"] = True 

1736 probe = fixations[keys].astype(str) 

1737 merged = probe.merge(present, on=keys, how="left") 

1738 keep = merged["_has_boxes"].notna().to_numpy(dtype=bool) 

1739 if not keep.all(): 

1740 dropped = fixations.loc[~keep, ["trial_id", "page"]].drop_duplicates() 

1741 _LOGGER.warning( 

1742 "MultiplEYE: dropped %d fixation(s) on %d screen(s) with no word " 

1743 "boxes (e.g. %s).", 

1744 int((~keep).sum()), 

1745 len(dropped), 

1746 ", ".join(f"{t}/{p}" for t, p in dropped.head(3).to_numpy()), 

1747 ) 

1748 return fixations.loc[keep]

1751def _multipleye_apply_screen_order( 

1752 words: pd.DataFrame, fixations: pd.DataFrame, *, by_onset: bool 

1753) -> tuple[pd.DataFrame, pd.DataFrame]: 

1754 """Stamp ``screen_index`` (+ the per-screen clock) on both frames. 

1755 

1756 Ranks only the **included** screens, so the indices stay a contiguous 1..N 

1757 once the rating / difficulty screens are gone. Per-reader word boxes take the 

1758 reader's own ranking; stimulus-level boxes (reading pages only, no 

1759 participant column to key on) take the page number, which is the same order 

1760 for every reader and — crucially — the same value the fixations get, so

1766 pd.to_numeric(fixations["onset"], errors="coerce") - fixations["_screen_onset"]

1769 if words.empty or "page" not in words.columns: 

1770 return words, fixations 

1771 if "participant_id" in words.columns: 

1772 words = words.merge(order.drop(columns=["_screen_onset"]), on=keys, how="left") 

1773 else: 

1774 numbers = words["page"].map(_multipleye_page_number) 

1775 if numbers.notna().all():

1908def multipleye_raw_frames( 

1909 root, 

1910 *, 

1911 sessions: Iterable[str] | None = None, 

1912 stimuli: Iterable[str] | None = None, 

1913 fixation_source: str = "scanpaths", 

1914 attach_reading_measures: bool = True, 

1915 include_question_screens: bool = True, 

1916) -> tuple[pd.DataFrame, pd.DataFrame]: 

1917 """Raw (pre-normalization) MultiplEYE ``(words, fixations)`` frames. 

1918 

1919 Same inputs as :func:`load_multipleye`, but returns the frames *before* 

1920 schema normalization — for callers that run their own auto-detection / 

1921 column mapping (e.g. the Streamlit app's MultiplEYE data source). Fixations 

1922 carry parsed ``participant_id`` (the session), ``trial_id`` (the stimulus), 

1923 the screen (``page`` + ``screen_index`` + ``screen_kind``), pixel 

1924 ``location_x/y``, the trial-level facets (``genre`` / ``session`` / 

1925 ``is_practice`` / ``trial_num``), and any reader metadata 

1926 (``pp_*`` from participant_data.csv), comprehension questions, and stimulus 

1927 image path that the corpus ships. 

1928 

1929 Word boxes are stimulus-level (no participant → broadcast) *unless* 

1930 ``reading_measures/`` exists and ``attach_reading_measures`` is on, or 

1931 question screens are included — in which case the boxes are emitted **per 

1932 reader**, with the corpus's pre-aggregated reading measures merged in as 

1933 ``IA_*`` columns (the app then prefers them over recomputed metrics) and the 

1934 reader's own question-screen boxes appended. ``include_question_screens`` 

1935 (on by default) opts out of the comprehension-question screens; they are 

1936 also skipped when the answer-layout version table is missing, since guessing 

1937 a layout would draw entirely plausible boxes in the wrong places. Use 

1938 :func:`load_multipleye` for normalized frames. 

1939 """ 

1940 root = Path(root) 

1941 if fixation_source not in MULTIPLEYE_FIXATION_SOURCES: 

1942 raise ValueError( 

1943 f"fixation_source must be one of {list(MULTIPLEYE_FIXATION_SOURCES)}, " 

1944 f"got {fixation_source!r}" 

1945 ) 

1946 if not (root / fixation_source).is_dir(): 

1947 # Fall back to whichever per-trial source folder exists. 

1948 alt = next( 

1949 (s for s in MULTIPLEYE_FIXATION_SOURCES if (root / s).is_dir()), None 

1950 ) 

1951 if alt is None: 

1952 raise FileNotFoundError( 

1953 f"No MultiplEYE fixation data under {root} — expected a " 

1954 f"'scanpaths' or 'fixations' folder of per-session subfolders." 

1955 ) 

1956 fixation_source = alt 

1957 

1958 fixations = _multipleye_fixations( 

1959 root, 

1960 fixation_source, 

1961 sessions, 

1962 stimuli, 

1963 include_question_screens=include_question_screens, 

1964 ) 

1965 # Reader metadata (age/gender/languages…) merged onto every fixation row. 

1966 fixations = _merge_multipleye_participant_meta( 

1967 fixations, _multipleye_participant_meta(root) 

1968 ) 

1969 

1970 aoi_dir = _multipleye_aoi_dir(root) 

1971 stim_boxes = _multipleye_word_boxes(aoi_dir, fixations["stimulus"].unique()) 

1972 versions = _multipleye_read_layout_versions(root) 

1973 question_boxes = ( 

1974 _multipleye_question_word_boxes(aoi_dir, fixations, versions) 

1975 if include_question_screens 

1976 else pd.DataFrame() 

1977 ) 

1978 # Pre-aggregated reading measures → per-reader word boxes (skips the 

1979 # stimulus-level broadcast). Only when the corpus ships reading_measures/. 

1980 # Question screens force the same shape: their layout is reader-specific. 

1981 per_reader = attach_reading_measures and (root / "reading_measures").is_dir() 

1982 if per_reader or not question_boxes.empty: 

1983 rm = pd.DataFrame() 

1984 if per_reader: 

1985 namemap = {s.rsplit("_", 1)[0]: s for s in fixations["stimulus"].unique()} 

1986 rm = _multipleye_read_reading_measures(root, sessions, namemap) 

1987 words = _multipleye_words_per_reader(stim_boxes, rm, fixations, question_boxes) 

1988 else: 

1989 words = stim_boxes 

1990 

1991 # Screens we could not build boxes for would be orphans in harmonize_frames, 

1992 # so drop them (loudly) before the screen order is ranked — that keeps the 

1993 # index a contiguous 1..N over exactly the screens that survive. 

1994 fixations = _multipleye_drop_screens_without_boxes(words, fixations) 

1995 words, fixations = _multipleye_apply_screen_order( 

1996 words, fixations, by_onset="participant_id" in words.columns 

1997 ) 

1998 

1999 # Comprehension questions + stimulus images, stamped on both frames. 

2000 qpath = _multipleye_questions_path(root) 

2001 if qpath is not None: 

2002 qmap = _multipleye_questions_by_stimulus(qpath) 

2003 words = _multipleye_stamp_questions(words, qmap) 

2004 fixations = _multipleye_stamp_questions(fixations, qmap) 

2005 image = _multipleye_image_dir(root) 

2006 if image is not None: 

2007 image_dir, lang = image 

2008 words = _multipleye_stamp_image_path(words, image_dir, lang) 

2009 fixations = _multipleye_stamp_image_path(fixations, image_dir, lang) 

2010 question_image = _multipleye_question_image_dir(root) 

2011 if question_image is not None: 

2012 image_dir, lang = question_image 

2013 words = _multipleye_stamp_question_image_path(words, image_dir, lang, versions) 

2014 fixations = _multipleye_stamp_question_image_path( 

2015 fixations, image_dir, lang, versions 

2016 ) 

2017 # Reading typeface (size + family) from the stimulus config → the app renders 

2018 # the text true-to-scale at the exact font the images were drawn with. 

2019 font_px, font_family = _multipleye_font_config(root) 

2020 words = _multipleye_stamp_font(words, font_px, font_family) 

2021 fixations = _multipleye_stamp_font(fixations, font_px, font_family) 

2022 return words, fixations

2028 sessions: Iterable[str] | None = None, 

2029 stimuli: Iterable[str] | None = None,
