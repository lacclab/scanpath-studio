1"""Downloads from the desktop app's own window open a Save-as dialog. 

2 

3The desktop app opens in a Chrome / Edge ``--app`` window. Current Chromium 

4reports a download in the toolbar's download bubble, and an app window has no 

5toolbar, so a download there would either save silently or show up in some 

6other browser window: clicking ⬇ Download GIF (or HTML, or anything else) 

7would look like it did nothing. 

8 

9Every download in the app ends in a click on an ``<a download>`` the page made 

10— ``st.download_button`` on a ``/media/…`` URL, the figure's PNG/SVG button 

11(`tabs._image_download_component`) on a ``blob:`` URL. Inside an app window 

12(``display-mode: standalone``) this takes that click instead: it asks the 

13browser's own Save-as dialog (``showSaveFilePicker``) where the file goes, 

14streams the file there and says so. A cancelled dialog saves nothing. Where 

15the dialog can't open (no user gesture left, an older browser), and in every 

16ordinary browser tab, the click is the browser's own, unchanged. 

17 

18Browser-only, like `color_picker_commit`: no session state and no setting, so 

19nothing to expose on the deep link / CLI / headless API. Installed once into 

20the parent document for the reason given there. 

21""" 

22 

23from __future__ import annotations 

24 

25import json 

26 

27from scanpath_studio.html_embed import embed_html_iframe 

28 

29SCRIPT_ID = "sps-save-dialog" 

30 

31# Runs in the parent document, where Streamlit and the v2 components make their 

32# download links, so patching the prototype's click reaches all of them. 

33_PARENT_JS = r""" 

34(function () { 

35 if (typeof window.showSaveFilePicker !== "function") return; 

36 const appWindow = () => window.matchMedia("(display-mode: standalone)").matches; 

37 const nativeClick = HTMLAnchorElement.prototype.click; 

38 

39 const nameFrom = (disposition) => { 

40 if (!disposition) return ""; 

41 const star = /filename\*\s*=\s*[^']*''([^;]+)/i.exec(disposition); 

42 if (star) { 

43 try { return decodeURIComponent(star[1].trim()); } catch (_e) { /* fall through */ } 

44 } 

45 const plain = /filename\s*=\s*"([^"]*)"|filename\s*=\s*([^;]+)/i.exec(disposition); 

46 return plain ? (plain[1] || plain[2] || "").trim() : ""; 

47 }; 

48 

49 const toast = (text, failed) => { 

50 const note = document.createElement("div"); 

51 note.textContent = text; 

52 note.setAttribute("role", "status"); 

53 note.style.cssText = [ 

54 "position:fixed", "right:1rem", "bottom:1rem", "z-index:1000001", 

55 "max-width:28rem", "padding:0.6rem 0.9rem", "border-radius:0.5rem", 

56 "font:14px/1.4 'Source Sans Pro',sans-serif", "color:#fff", 

57 "box-shadow:0 4px 14px rgba(0,0,0,0.25)", 

58 "background:" + (failed ? "#b3261e" : "#1e6e43"), 

59 ].join(";"); 

60 document.body.appendChild(note); 

61 setTimeout(() => note.remove(), failed ? 8000 : 4000); 

62 }; 

63 

64 async function save(link, url) { 

65 let name = link.getAttribute("download") || ""; 

66 if (!name && url.protocol !== "blob:") { 

67 const head = await fetch(url, { method: "HEAD" }); 

68 name = nameFrom(head.headers.get("Content-Disposition")); 

69 } 

70 if (!name) name = decodeURIComponent(url.pathname.split("/").pop() || "download"); 

71 // Throws before anything is written: no gesture left, or cancelled. 

72 const handle = await window.showSaveFilePicker({ suggestedName: name }); 

73 try { 

74 const response = await fetch(url); 

75 if (!response.ok) throw new Error("the app answered " + response.status); 

76 await response.body.pipeTo(await handle.createWritable()); 

77 } catch (err) { 

78 toast("Couldn't save " + handle.name + " (" + ((err && err.message) || err) + "). Try again.", true); 

79 return; 

80 } 

81 toast("Saved " + handle.name); 

82 } 

83 

84 HTMLAnchorElement.prototype.click = function () { 

85 if (!this.hasAttribute("download") || !appWindow()) return nativeClick.call(this); 

86 let url; 

87 try { url = new URL(this.href, window.location.href); } catch (_e) { return nativeClick.call(this); } 

88 if (url.protocol !== "blob:" && url.origin !== window.location.origin) { 

89 return nativeClick.call(this); 

90 } 

91 const link = this; 

92 save(link, url).catch((err) => { 

93 if (err && err.name === "AbortError") return; // the dialog was cancelled 

94 nativeClick.call(link); // the dialog could not open: the browser's own download 

95 }); 

96 }; 

97})(); 

98""" 

99 

100_JS = """ 

101(function () { 

102 const doc = window.parent.document; 

103 if (!doc.head || doc.getElementById(__ID__)) return; 

104 const script = doc.createElement("script"); 

105 script.id = __ID__; 

106 script.textContent = __BODY__; 

107 doc.head.appendChild(script); 

108})(); 

109""" 

110 

111 

112def save_dialog_script() -> str: 

113 """The same-origin ``<script>`` that installs the hook once.""" 

114 body = _JS.replace("__ID__", json.dumps(SCRIPT_ID)).replace( 

115 "__BODY__", json.dumps(_PARENT_JS) 

116 ) 

117 return f"<script>{body}</script>" 

118 

119 

120def render_save_dialog() -> None: 

121 """Install the hook (a no-op in the browser after the first run).""" 

122 embed_html_iframe(save_dialog_script(), height=0)
